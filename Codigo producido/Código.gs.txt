// Caché en memoria para optimizar ejecuciones concurrentes sin exceder cuotas de lectura
var _CACHE_PARAMETROS_MEMORIA = null;

/**
 * Consulta la hoja "Parametros" en el archivo Parametros_Viaticos2.0 y retorna
 * un mapa clave-valor con todos los IDs de bases de datos y carpetas registradas.
 * Utiliza CacheService y caché en memoria para alto rendimiento.
 */
function obtenerMapaParametros(forzarRecarga) {
  if (!forzarRecarga && _CACHE_PARAMETROS_MEMORIA) {
    return _CACHE_PARAMETROS_MEMORIA;
  }

  try {
    var cache = CacheService.getScriptCache();
    if (!forzarRecarga) {
      var cached = cache.get("MAPA_PARAMETROS_VIATICOS");
      if (cached) {
        _CACHE_PARAMETROS_MEMORIA = JSON.parse(cached);
        return _CACHE_PARAMETROS_MEMORIA;
      }
    }

    var ssParametros = SpreadsheetApp.openById(ID_DB_PARAMETROS);
    var sheet = ssParametros.getSheetByName("Parametros") || ssParametros.getSheets()[0];
    var data = sheet.getDataRange().getValues();
    
    if (data.length < 2) {
      Logger.log("Advertencia: La hoja de Parámetros está vacía.");
      return {};
    }

    var headers = data[0];
    var colClave = -1;
    var colValor = -1;

    for (var h = 0; h < headers.length; h++) {
      var headerText = headers[h].toString().trim().toUpperCase();
      if (headerText === "CLAVE" || headerText === "KEY" || headerText === "PARAMETRO") colClave = h;
      if (headerText === "VALOR" || headerText === "VALUE" || headerText === "ID") colValor = h;
    }

    if (colClave === -1) colClave = 0; // Default columna A
    if (colValor === -1) colValor = 1; // Default columna B

    var mapa = {};
    for (var r = 1; r < data.length; r++) {
      var clave = (data[r][colClave] || "").toString().trim();
      var valor = (data[r][colValor] || "").toString().trim();
      if (clave) {
        mapa[clave] = valor;
      }
    }

    _CACHE_PARAMETROS_MEMORIA = mapa;
    try {
      cache.put("MAPA_PARAMETROS_VIATICOS", JSON.stringify(mapa), 21600); // 6 horas
    } catch (eCache) {
      Logger.log("Error guardando en CacheService: " + eCache.message);
    }

    return mapa;
  } catch (error) {
    Logger.log("Error al consultar Parametros_Viaticos2.0: " + error.toString());
    return _CACHE_PARAMETROS_MEMORIA || {};
  }
}

/**
 * Obtiene el valor de un parámetro específico por su clave (ej: 'ID_DB_MAESTRO').
 */
function obtenerParametro(nombreClave) {
  var mapa = obtenerMapaParametros(false);
  return mapa[nombreClave] || null;
}

/**
 * Función utilitaria para invalidar la caché y forzar la relectura de Parametros_Viaticos2.0
 */
function recargarParametros() {
  _CACHE_PARAMETROS_MEMORIA = null;
  var cache = CacheService.getScriptCache();
  if (cache) cache.remove("MAPA_PARAMETROS_VIATICOS");
  return obtenerMapaParametros(true);
}

/**
 * Normaliza textos de encabezado para mapeos insensibles a mayúsculas, espacios y caracteres especiales.
 */
function normalizarTextoClave(txt) {
  if (!txt) return "";
  return txt.toString().toLowerCase()
    .normalize("NFD").replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]/g, "");
}

/**
 * Obtiene el índice (0-indexed) de una columna con resiliencia total y fallbacks.
 */
function obtenerIndiceColumna(colMap, key, defaultColIdx) {
  if (!colMap) return defaultColIdx !== undefined ? defaultColIdx : 0;
  if (colMap[key] !== undefined) return colMap[key];
  var norm = normalizarTextoClave(key);
  if (colMap[norm] !== undefined) return colMap[norm];
  if (_CANONICAL_COL_INDEX && _CANONICAL_COL_INDEX[key] !== undefined) {
    return _CANONICAL_COL_INDEX[key];
  }
  return defaultColIdx !== undefined ? defaultColIdx : 0;
}

/**
 * Extrae con resiliencia el primer valor no vacío de una fila para una lista de claves candidatas.
 * Formatea objetos Date a "dd/MM/yyyy" automáticamente para evitar desfases de zona horaria.
 */
function obtenerColVal(row, colMap, keys, defaultVal) {
  if (!row || !colMap) return defaultVal !== undefined ? defaultVal : "";
  for (var i = 0; i < keys.length; i++) {
    var k = keys[i];
    var idx = colMap[k];
    if (idx === undefined) {
      var norm = normalizarTextoClave(k);
      idx = colMap[norm];
    }
    if (idx === undefined && _CANONICAL_COL_INDEX && _CANONICAL_COL_INDEX[k] !== undefined) {
      idx = _CANONICAL_COL_INDEX[k];
    }
    if (idx !== undefined && row[idx] !== undefined && row[idx] !== null && row[idx] !== "") {
      var val = row[idx];
      if (val instanceof Date) {
        if (!isNaN(val.getTime())) {
          return Utilities.formatDate(val, Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy");
        }
      }
      return val;
    }
  }
  return defaultVal !== undefined ? defaultVal : "";
}

/**
 * Escribe con resiliencia y seguridad un valor en la columna indicada de una hoja de cálculo.
 */
function establecerColVal(sheet, fila, colMap, key, val, defaultColIdx) {
  if (!sheet || !fila) return;
  var idx = obtenerIndiceColumna(colMap, key, defaultColIdx);
  if (idx !== undefined && !isNaN(idx) && idx >= 0) {
    sheet.getRange(fila, idx + 1).setValue(val);
  }
}

/**
 * Genera una respuesta estructurada y consistente para el cliente frontend.
 * @param {boolean} success - Indica si la operación concluyó exitosamente
 * @param {*} data - Carga útil de datos de retorno
 * @param {string} mensaje - Mensaje legible para el usuario
 * @param {Object} [extra] - Metadatos adicionales opcionales
 */
function crearRespuestaAPI(success, data, mensaje, extra) {
  var res = {
    success: !!success,
    status: success ? 'OK' : 'ERROR',
    data: data !== undefined ? data : null,
    mensaje: mensaje || (success ? "Operación exitosa" : "Error en la operación")
  };
  if (extra && typeof extra === 'object') {
    for (var k in extra) {
      if (extra.hasOwnProperty(k)) res[k] = extra[k];
    }
  }
  return res;
}

/**
 * Combina un JSON previo de archivos adjuntos con nuevos archivos adjuntos en modo append (sin sobreescribir).
 * @param {string|Array} adjuntosPreviosJSON
 * @param {Array} nuevosArchivosArray
 * @returns {string} JSON resultante consolidado
 */
function combinarArchivosAdjuntosJSON(adjuntosPreviosJSON, nuevosArchivosArray) {
  var lista = [];
  if (adjuntosPreviosJSON) {
    if (typeof adjuntosPreviosJSON === 'string') {
      try {
        var parsed = JSON.parse(adjuntosPreviosJSON);
        if (Array.isArray(parsed)) lista = parsed;
      } catch (e) {
        lista = [];
      }
    } else if (Array.isArray(adjuntosPreviosJSON)) {
      lista = adjuntosPreviosJSON.slice();
    }
  }

  if (nuevosArchivosArray && Array.isArray(nuevosArchivosArray)) {
    for (var i = 0; i < nuevosArchivosArray.length; i++) {
      if (nuevosArchivosArray[i]) {
        lista.push(nuevosArchivosArray[i]);
      }
    }
  }

  return JSON.stringify(lista);
}

/**
 * Restituye el saldo devuelto de un viático (MontoReintegro) al presupuesto disponible en DimDisponible y lo descuenta de DimConsumo.
 * Se ejecuta tras la aprobación en E3.1 cuando TipoCierre == 'Reintegro y cierre'.
 * @param {string} idSolicitud - ID de la solicitud liquidada
 * @param {number} montoReintegro - Valor monetario devuelto
 * @param {string|Date} fechaCierreE - Fecha de cierre emitida por Compras
 * @returns {Object} Respuesta API con el estado de la operación
 */
function restituirPresupuestoPorReintegro(idSolicitud, montoReintegro, fechaCierreE) {
  var monto = parseFloat(montoReintegro);
  if (isNaN(monto) || monto <= 0) {
    return crearRespuestaAPI(true, null, "No hay monto de reintegro que devolver (monto <= 0).");
  }

  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); // 10 segundos de espera concurrente
  } catch (eLock) {
    return crearRespuestaAPI(false, null, "El sistema de presupuestos está ocupado. Por favor intente de nuevo.");
  }

  try {
    var idDbMaestro = obtenerParametro("ID_DB_MAESTRO");
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONAL");
    if (!idDbMaestro || !idDbTransaccional) {
      return crearRespuestaAPI(false, null, "Faltan parámetros de base de datos en Parametros_Viaticos2.0.");
    }

    var ssTrans = SpreadsheetApp.openById(idDbTransaccional);
    var sheetTrans = ssTrans.getSheetByName("DimTransaccional");
    if (!sheetTrans) {
      return crearRespuestaAPI(false, null, "No se encontró la hoja DimTransaccional.");
    }

    var transData = sheetTrans.getDataRange().getValues();
    var headersTrans = transData[0];
    var colId = -1, colCC = -1, colRC = -1;

    for (var h = 0; h < headersTrans.length; h++) {
      var ht = headersTrans[h].toString().trim().toUpperCase();
      if (ht === "ID_SOLICITUD" || ht === "ID") colId = h;
      if (ht === "CENTROCOSTO" || ht === "CENTRO_COSTO") colCC = h;
      if (ht === "RUBROCONTABLE" || ht === "RUBRO_CONTABLE" || ht === "CODIGORUBRO") colRC = h;
    }

    var centroCosto = "", rubroContable = "";
    for (var r = 1; r < transData.length; r++) {
      if (transData[r][colId].toString().trim() === idSolicitud.toString().trim()) {
        centroCosto = colCC !== -1 ? transData[r][colCC].toString().trim() : "";
        rubroContable = colRC !== -1 ? transData[r][colRC].toString().trim() : "";
        break;
      }
    }

    if (!centroCosto || !rubroContable) {
      return crearRespuestaAPI(false, null, "No se encontraron Centro de Costo y Rubro Contable para la solicitud " + idSolicitud);
    }

    // Determinar el mes aplicable
    var fecha = fechaCierreE ? new Date(fechaCierreE) : new Date();
    if (isNaN(fecha.getTime())) fecha = new Date();
    var idxMes = fecha.getMonth(); // 0-11
    var nombreMes = MESES_ANIO[idxMes];

    var ssMaestro = SpreadsheetApp.openById(idDbMaestro);
    var sheetDisponible = ssMaestro.getSheetByName("DimDisponible");
    var sheetConsumo = ssMaestro.getSheetByName("DimConsumo");

    if (!sheetDisponible || !sheetConsumo) {
      return crearRespuestaAPI(false, null, "No se encontraron las pestañas DimDisponible y/o DimConsumo en DB Maestro.");
    }

    // Actualizar DimDisponible (Suma del monto a reintegrar)
    var dispData = sheetDisponible.getDataRange().getValues();
    var dispHeaders = dispData[0];
    var colMesDisp = -1, colCCDisp = -1, colRCDisp = -1;

    for (var cd = 0; cd < dispHeaders.length; cd++) {
      var hdText = dispHeaders[cd].toString().trim().toUpperCase();
      if (hdText === "CENTROCOSTO" || hdText === "CENTRO_COSTO") colCCDisp = cd;
      if (hdText === "RUBROCONTABLE" || hdText === "RUBRO_CONTABLE") colRCDisp = cd;
      if (hdText === nombreMes.toUpperCase()) colMesDisp = cd;
    }

    if (colMesDisp !== -1 && colCCDisp !== -1 && colRCDisp !== -1) {
      for (var rd = 1; rd < dispData.length; rd++) {
        var ccRow = dispData[rd][colCCDisp].toString().trim();
        var rcRow = dispData[rd][colRCDisp].toString().trim();
        if (ccRow === centroCosto && rcRow === rubroContable) {
          var saldoActual = parseFloat(dispData[rd][colMesDisp]) || 0;
          var nuevoSaldo = saldoActual + monto;
          sheetDisponible.getRange(rd + 1, colMesDisp + 1).setValue(nuevoSaldo);
          break;
        }
      }
    }

    // Actualizar DimConsumo (Resta del monto a reintegrar)
    var consData = sheetConsumo.getDataRange().getValues();
    var consHeaders = consData[0];
    var colMesCons = -1, colCCCons = -1, colRSCons = -1;

    for (var cs = 0; cs < consHeaders.length; cs++) {
      var hsText = consHeaders[cs].toString().trim().toUpperCase();
      if (hsText === "CENTROCOSTO" || hsText === "CENTRO_COSTO") colCCCons = cs;
      if (hsText === "RUBROCONTABLE" || hsText === "RUBRO_CONTABLE") colRSCons = cs;
      if (hsText === nombreMes.toUpperCase()) colMesCons = cs;
    }

    if (colMesCons !== -1 && colCCCons !== -1 && colRSCons !== -1) {
      for (var rs = 1; rs < consData.length; rs++) {
        var ccRowS = consData[rs][colCCCons].toString().trim();
        var rcRowS = consData[rs][colRSCons].toString().trim();
        if (ccRowS === centroCosto && rcRowS === rubroContable) {
          var consumoActual = parseFloat(consData[rs][colMesCons]) || 0;
          var nuevoConsumo = Math.max(0, consumoActual - monto);
          sheetConsumo.getRange(rs + 1, colMesCons + 1).setValue(nuevoConsumo);
          break;
        }
      }
    }

    SpreadsheetApp.flush();
    return crearRespuestaAPI(true, {
      idSolicitud: idSolicitud,
      montoReintegro: monto,
      mes: nombreMes,
      centroCosto: centroCosto,
      rubroContable: rubroContable
    }, "Devolución presupuestaria por reintegro procesada con éxito.");

  } catch (error) {
    Logger.log("Error en restituirPresupuestoPorReintegro: " + error.toString());
    return crearRespuestaAPI(false, null, "Error al procesar devolución de presupuesto: " + error.toString());
  } finally {
    lock.releaseLock();
  }
}

function obtenerUsuarioGoogle() {
  var email = Session.getActiveUser().getEmail() || "";
  var nombre = email ? email.split('@')[0].replace('.', ' ') : "Usuario";
  nombre = nombre.replace(/\b\w/g, function(l) { return l.toUpperCase(); });
  
  var iniciales = nombre.split(' ').map(function(n) { return n.charAt(0); }).join('').substring(0, 2).toUpperCase() || "U";
  var foto = "https://placehold.co/150x150/001f5c/ffffff?text=" + iniciales + "&font=segoe";

  try {
    if (typeof People !== 'undefined' && People.People && email) {
      var respuesta = People.People.searchDirectoryPeople({
        query: email,
        readMask: 'names,photos',
        sources: ['DIRECTORY_SOURCE_TYPE_DOMAIN_PROFILE']
      });

      if (respuesta.people && respuesta.people.length > 0) {
        var persona = respuesta.people[0];
        if (persona.names && persona.names.length > 0 && persona.names[0].displayName) {
          nombre = persona.names[0].displayName;
        }
        if (persona.photos && persona.photos.length > 0 && persona.photos[0].url) {
          foto = persona.photos[0].url;
        }
      }
    }
  } catch (e) {
    Logger.log("People API fallback: " + e.message);
  }
  
  return {
    nombre: nombre,
    correo: email,
    foto: foto
  };
}

function validarDimUsuarios(correoUsuario) {
  try {
    var idDbMaestro = obtenerParametro("ID_DB_MAESTRO");
    if (!idDbMaestro) return { status: 'ERROR', mensaje: 'No se configuró ID_DB_MAESTRO en Parametros_Viaticos2.0.' };
    
    var ssMaestro = SpreadsheetApp.openById(idDbMaestro);
    var sheetUsuarios = ssMaestro.getSheetByName("DimUsuarios");
    if (!sheetUsuarios) return { status: 'ERROR', mensaje: 'No se encontró la pestaña "DimUsuarios".' };
    
    var data = sheetUsuarios.getDataRange().getValues();
    if (data.length < 2) return { status: 'NO_EXISTE', accesoViaticos: null };
    
    var headers = data[0];
    var colCorreo = -1, colAcceso = -1, colRol = -1;
    
    for (var i = 0; i < headers.length; i++) {
      var h = headers[i].toString().toLowerCase().trim();
      if (h === "correousuario" || h === "correo" || h === "email") colCorreo = i;
      if (h === "accesoviaticos" || h === "acceso") colAcceso = i;
      if (h === "rolusuario_viaticos" || h === "rol") colRol = i;
    }
    
    if (colCorreo === -1) return { status: 'ERROR', mensaje: 'Falta columna CorreoUsuario en DimUsuarios.' };
    
    var correoBuscado = (correoUsuario || "").toString().toLowerCase().trim();
    
    for (var r = 1; r < data.length; r++) {
      var correoFila = data[r][colCorreo].toString().toLowerCase().trim();
      if (correoFila === correoBuscado) {
        var accesoVal = colAcceso !== -1 ? data[r][colAcceso].toString().toUpperCase().trim() : "SI";
        var rolVal = colRol !== -1 ? data[r][colRol].toString().toUpperCase().trim() : "SOLICITANTE";
        
        var filaUsuario = {};
        for (var c = 0; c < headers.length; c++) {
          filaUsuario[headers[c].toString().trim()] = data[r][c];
        }
        
        var perfilGoogle = obtenerUsuarioGoogle();
        filaUsuario["NombreGoogle"] = perfilGoogle.nombre;
        filaUsuario["CorreoGoogle"] = perfilGoogle.correo;
        filaUsuario["FotoGoogle"] = perfilGoogle.foto;
        filaUsuario["RolUsuario_Viaticos"] = rolVal || "SOLICITANTE";

        var tieneAcceso = (accesoVal === "SI" || accesoVal === "" || accesoVal === "TRUE");

        return { 
          status: 'EXISTE', 
          accesoViaticos: tieneAcceso ? "SI" : "NO", 
          datosFila: filaUsuario 
        };
      }
    }
    
    return { status: 'NO_EXISTE', accesoViaticos: null };
  } catch (error) {
    return { status: 'ERROR', mensaje: 'Error al validar usuario: ' + error.toString() };
  }
}

function obtenerUsuariosPorCentroCosto(centroCosto) {

/**
 * Guarda una lista de archivos en base64 en la carpeta designada en Drive y devuelve la lista de objetos con LinkArchivo.
 * Soporta subcarpetas por correlativo ("Archivos - " + idSolicitud), mapeo flexible de propiedades y fallbacks seguros.
 */
function guardarArchivosEnDrive(idSolicitud, listaArchivos) {
  var archivosGuardados = [];
  if (!listaArchivos) return archivosGuardados;
  if (!Array.isArray(listaArchivos)) listaArchivos = [listaArchivos];
  if (listaArchivos.length === 0) return archivosGuardados;
  
  try {
    var idCarpetaRaiz = obtenerParametro("ID_CARPETA_ADJUNTOS") || 
                        obtenerParametro("ID_CARPETA_COMPROBANTES") || 
                        obtenerParametro("ID_CARPETA_ARCHIVOS") || 
                        obtenerParametro("ID_CARPETA_DRIVE");
    var carpetaRaiz = null;
    if (idCarpetaRaiz) {
      try {
        carpetaRaiz = DriveApp.getFolderById(idCarpetaRaiz);
      } catch (eFolder) {
        Logger.log("Aviso: No se pudo acceder a carpeta por ID " + idCarpetaRaiz + ": " + eFolder.toString());
      }
    }
    if (!carpetaRaiz) {
      try {
        carpetaRaiz = DriveApp.getRootFolder();
      } catch (eRoot) {
        Logger.log("Aviso: No se pudo acceder a root folder de Drive: " + eRoot.toString());
      }
    }
    
    var subcarpeta = carpetaRaiz;
    if (carpetaRaiz && idSolicitud) {
      try {
        var strId = idSolicitud.toString();
        var nombreSubcarpeta = (strId.indexOf("Archivos - ") === 0) ? strId : ("Archivos - " + strId);
        var carpetasExistentes = carpetaRaiz.getFoldersByName(nombreSubcarpeta);
        subcarpeta = carpetasExistentes.hasNext() ? carpetasExistentes.next() : carpetaRaiz.createFolder(nombreSubcarpeta);
      } catch (eSub) {
        Logger.log("Aviso: No se pudo crear subcarpeta específica, se usará carpeta contenedora: " + eSub.toString());
        subcarpeta = carpetaRaiz;
      }
    }
    
    for (var i = 0; i < listaArchivos.length; i++) {
      var item = listaArchivos[i];
      if (!item) continue;
      
      var base64Raw = item.contenidoBase64 || item.ContenidoBase64 || item.dataUrl || item.DataUrl || item.base64 || item.Base64 || "";
      var nombreArchivo = item.nombre || item.NombreArchivo || item.name || ("Adjunto_" + idSolicitud + "_" + (i + 1));
      var tipoMime = item.tipo || item.TipoArchivo || item.type || item.mimeType || "application/octet-stream";
      var linkExistente = item.LinkArchivo || item.linkArchivo || item.url || item.Url || "";
      var idExistente = item.IdArchivo || item.idArchivo || item.id || "";
      
      if (base64Raw && subcarpeta) {
        try {
          var base64Limpio = base64Raw.indexOf('base64,') > -1 ? base64Raw.split('base64,')[1] : (base64Raw.indexOf(',') > -1 ? base64Raw.split(',')[1] : base64Raw);
          base64Limpio = base64Limpio.trim();
          var bytes = Utilities.base64Decode(base64Limpio);
          var blob = Utilities.newBlob(bytes, tipoMime, nombreArchivo);
          var archivoDrive = subcarpeta.createFile(blob);
          
          try {
            archivoDrive.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          } catch (eShare) {
            // Continuar si setSharing falla en dominios restringidos
          }
          
          archivosGuardados.push({
            NombreArchivo: nombreArchivo,
            TipoArchivo: tipoMime,
            LinkArchivo: archivoDrive.getUrl(),
            IdArchivo: archivoDrive.getId()
          });
        } catch (eFile) {
          Logger.log("Error al procesar archivo en Drive: " + eFile.toString());
          archivosGuardados.push({
            NombreArchivo: nombreArchivo,
            TipoArchivo: tipoMime,
            LinkArchivo: linkExistente || "",
            IdArchivo: idExistente || ""
          });
        }
      } else if (linkExistente) {
        archivosGuardados.push({
          NombreArchivo: nombreArchivo,
          TipoArchivo: tipoMime,
          LinkArchivo: linkExistente,
          IdArchivo: idExistente || ""
        });
      } else {
        archivosGuardados.push({
          NombreArchivo: nombreArchivo,
          TipoArchivo: tipoMime,
          LinkArchivo: "",
          IdArchivo: ""
        });
      }
    }
  } catch (err) {
    Logger.log("Error al subir archivos a Drive: " + err.toString());
    for (var j = 0; j < listaArchivos.length; j++) {
      var arch = listaArchivos[j];
      if (arch) {
        archivosGuardados.push({
          NombreArchivo: arch.nombre || arch.NombreArchivo || arch.name || "Archivo",
          TipoArchivo: arch.tipo || arch.TipoArchivo || arch.type || "application/octet-stream",
          LinkArchivo: arch.url || arch.LinkArchivo || arch.linkArchivo || "",
          IdArchivo: arch.IdArchivo || arch.idArchivo || ""
        });
      }
    }
  }
  
  return archivosGuardados;
}

// 7. MOTOR DE VALIDACIÓN Y AFECTACIÓN PRESUPUESTARIA
// --------------------------------------------------------------------------
function validarYAfectarPresupuesto(centroCostoNum, rubroCodigo, fechaSolicitudStr, monto) {
  try {
    var idDbPresupuesto = obtenerParametro("ID_DB_PRESUPUESTO");
    if (!idDbPresupuesto) {
      return { status: 'ERROR', mensaje: 'No se encontró ID_DB_PRESUPUESTO en Parametros_Viaticos2.0.' };
    }
    var ssPresupuesto = SpreadsheetApp.openById(idDbPresupuesto);
    
    var sheetDisponible = ssPresupuesto.getSheetByName("DimDisponible");
    var sheetConsumo = ssPresupuesto.getSheetByName("DimConsumo");
    
    if (!sheetDisponible || !sheetConsumo) {
      return { status: 'ERROR', mensaje: 'Faltan tablas DimDisponible o DimConsumo en Presupuesto.' };
    }
    
    // Determinar nombre del mes a partir de FechaSolicitud
    var fecha = fechaSolicitudStr ? parsearFechaObj(fechaSolicitudStr) : new Date();
    if (isNaN(fecha.getTime())) fecha = new Date();
    var nombreMes = MESES_ANIO[fecha.getMonth()] || "Enero";
    
    // Buscar coordenada de columna para el mes
    var headersDisp = sheetDisponible.getRange(1, 1, 1, sheetDisponible.getLastColumn()).getValues()[0];
    var colIndexMes = -1;
    for (var c = 0; c < headersDisp.length; c++) {
      if (headersDisp[c].toString().trim().toUpperCase() === nombreMes.toUpperCase()) {
        colIndexMes = c + 1;
        break;
      }
    }
    
    if (colIndexMes === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la columna del mes ' + nombreMes + ' en DimDisponible.' };
    }
    
    // Construir llave de búsqueda CC-RC
    var ccLimpio = (centroCostoNum || "").toString().trim();
    var rcLimpio = (rubroCodigo || "").toString().trim();
    var llaveCompleta = ccLimpio + "-" + rcLimpio;
    
    var datosDisp = sheetDisponible.getDataRange().getValues();
    var filaEncontrada = -1;
    
    for (var r = 1; r < datosDisp.length; r++) {
      var llaveFila = (datosDisp[r][0] || "").toString().trim();
      if (llaveFila === llaveCompleta || llaveFila.indexOf(ccLimpio + "-") === 0) {
        filaEncontrada = r + 1;
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return {
        status: 'OK',
        dentroPresupuesto: 'No',
        monto: monto,
        disponibleActual: 0,
        mensaje: 'No se encontró asignación presupuestaria para ' + llaveCompleta
      };
    }
    
    var disponibleActual = parseFloat(sheetDisponible.getRange(filaEncontrada, colIndexMes).getValue() || 0);
    var montoNum = parseFloat(monto || 0);
    
    if (disponibleActual >= montoNum) {
      // Hay presupuesto: Afectar saldos
      var nuevoDisponible = disponibleActual - montoNum;
      sheetDisponible.getRange(filaEncontrada, colIndexMes).setValue(nuevoDisponible);
      
      var consumoActual = parseFloat(sheetConsumo.getRange(filaEncontrada, colIndexMes).getValue() || 0);
      sheetConsumo.getRange(filaEncontrada, colIndexMes).setValue(consumoActual + montoNum);
      
      return {
        status: 'OK',
        dentroPresupuesto: 'Si',
        disponibleAnterior: disponibleActual,
        nuevoDisponible: nuevoDisponible
      };
    } else {
      // Fuera de presupuesto
      return {
        status: 'OK',
        dentroPresupuesto: 'No',
        disponibleActual: disponibleActual,
        monto: montoNum
      };
    }
    
  } catch (error) {
    return { status: 'ERROR', mensaje: 'Error al verificar presupuesto: ' + error.toString() };
  }
}

/**
 * Convierte cualquier formato de fecha de Google Sheets o string (dd/MM/yyyy, yyyy-MM-dd, ISO, Date) en un objeto Date válido.
 */
function parsearFechaObj(val) {
  if (!val) return new Date();
  if (val instanceof Date) return isNaN(val.getTime()) ? new Date() : val;
  var str = val.toString().trim();
  if (str === "") return new Date();
  
  if (str.indexOf('/') !== -1) {
    var partesEspacio = str.split(' ');
    var partesFecha = partesEspacio[0].split('/');
    if (partesFecha.length === 3) {
      var d = parseInt(partesFecha[0], 10);
      var m = parseInt(partesFecha[1], 10) - 1;
      var y = parseInt(partesFecha[2], 10);
      var dt = new Date(y, m, d);
      if (!isNaN(dt.getTime())) return dt;
    }
  }
  
  if (str.indexOf('-') !== -1) {
    var partesGuion = str.split('T')[0].split(' ')[0].split('-');
    if (partesGuion.length === 3) {
      var y = parseInt(partesGuion[0], 10);
      var m = parseInt(partesGuion[1], 10) - 1;
      var d = parseInt(partesGuion[2], 10);
      var dt = new Date(y, m, d);
      if (!isNaN(dt.getTime())) return dt;
    }
  }
  
  var parsed = new Date(str);
  return isNaN(parsed.getTime()) ? new Date() : parsed;
}

/**
 * Formatea cualquier valor de fecha a string legible 'dd/MM/yyyy' o 'dd/MM/yyyy HH:mm'.
 */
function formatearFechaTexto(val, incluirHora) {
  if (!val) return "";
  if (val instanceof Date) {
    if (isNaN(val.getTime())) return "";
    var formato = incluirHora ? "dd/MM/yyyy HH:mm" : "dd/MM/yyyy";
    return Utilities.formatDate(val, Session.getScriptTimeZone() || "GMT-6", formato);
  }
  var str = val.toString().trim();
  if (str === "") return "";
  
  // Si ya es un formato dd/MM/yyyy o dd/MM/yyyy HH:mm
  if (/^\d{2}\/\d{2}\/\d{4}/.test(str)) {
    return str;
  }
  
  var dt = parsearFechaObj(str);
  if (dt && !isNaN(dt.getTime())) {
    var formato = incluirHora ? "dd/MM/yyyy HH:mm" : "dd/MM/yyyy";
    return Utilities.formatDate(dt, Session.getScriptTimeZone() || "GMT-6", formato);
  }
  return str;
}

/**
 * Reversa la afectación presupuestaria: resta de DimConsumo y suma de vuelta a DimDisponible.
 */
function revertirAfectacionPresupuesto(centroCostoNum, rubroCodigo, fechaSolicitudStr, monto) {
  try {
    var idDbPresupuesto = obtenerParametro("ID_DB_PRESUPUESTO");
    if (!idDbPresupuesto) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_PRESUPUESTO en Parametros_Viaticos2.0.' };
    
    var ssPresupuesto = SpreadsheetApp.openById(idDbPresupuesto);
    var sheetDisponible = ssPresupuesto.getSheetByName("DimDisponible");
    var sheetConsumo = ssPresupuesto.getSheetByName("DimConsumo");
    
    if (!sheetDisponible || !sheetConsumo) {
      return { status: 'ERROR', mensaje: 'Faltan tablas DimDisponible o DimConsumo en Presupuesto.' };
    }
    
    var fecha = fechaSolicitudStr ? parsearFechaObj(fechaSolicitudStr) : new Date();
    var nombreMes = MESES_ANIO[fecha.getMonth()] || "Enero";
    
    var headersDisp = sheetDisponible.getRange(1, 1, 1, sheetDisponible.getLastColumn()).getValues()[0];
    var colIndexMes = -1;
    for (var c = 0; c < headersDisp.length; c++) {
      if (headersDisp[c].toString().trim().toUpperCase() === nombreMes.toUpperCase()) {
        colIndexMes = c + 1;
        break;
      }
    }
    
    if (colIndexMes === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la columna del mes ' + nombreMes + ' en DimDisponible.' };
    }
    
    var ccLimpio = (centroCostoNum || "").toString().trim();
    var rcLimpio = (rubroCodigo || "").toString().trim();
    var llaveCompleta = ccLimpio + "-" + rcLimpio;
    
    var datosDisp = sheetDisponible.getDataRange().getValues();
    var filaEncontrada = -1;
    
    for (var r = 1; r < datosDisp.length; r++) {
      var llaveFila = (datosDisp[r][0] || "").toString().trim();
      if (llaveFila === llaveCompleta || llaveFila.indexOf(ccLimpio + "-") === 0) {
        filaEncontrada = r + 1;
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró asignación presupuestaria para ' + llaveCompleta };
    }
    
    var montoNum = parseFloat(monto || 0);
    if (montoNum > 0) {
      var disponibleActual = parseFloat(sheetDisponible.getRange(filaEncontrada, colIndexMes).getValue() || 0);
      var consumoActual = parseFloat(sheetConsumo.getRange(filaEncontrada, colIndexMes).getValue() || 0);
      
      var nuevoDisponible = disponibleActual + montoNum;
      var nuevoConsumo = Math.max(0, consumoActual - montoNum);
      
      sheetDisponible.getRange(filaEncontrada, colIndexMes).setValue(nuevoDisponible);
      sheetConsumo.getRange(filaEncontrada, colIndexMes).setValue(nuevoConsumo);
      
      return { status: 'OK', liberado: true, nuevoDisponible: nuevoDisponible, nuevoConsumo: nuevoConsumo };
    }
    
    return { status: 'OK', liberado: true };
  } catch (error) {
    Logger.log("Error en revertirAfectacionPresupuesto: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al liberar presupuesto: ' + error.toString() };
  }
}

/**
 * Obtiene los correos de los usuarios que poseen un rol determinado (ej. 'EDITOR-COMPRAS').
 */
function obtenerCorreosPorRol(rolBuscado) {
  try {
    var idDbMaestro = obtenerParametro("ID_DB_MAESTRO");
    if (!idDbMaestro) return [];
    var ssMaestro = SpreadsheetApp.openById(idDbMaestro);
    var sheetUsuarios = ssMaestro.getSheetByName("DimUsuarios");
    if (!sheetUsuarios) return [];
    
    var data = sheetUsuarios.getDataRange().getValues();
    if (data.length < 2) return [];
    
    var headers = data[0];
    var colCorreo = -1, colRol = -1;
    for (var i = 0; i < headers.length; i++) {
      var h = headers[i].toString().toLowerCase().trim();
      if (h === "correousuario" || h === "correo" || h === "email") colCorreo = i;
      if (h === "rolusuario_viaticos" || h === "rol") colRol = i;
    }
    
    if (colCorreo === -1 || colRol === -1) return [];
    var rolUpper = (rolBuscado || "").toUpperCase().trim();
    var correos = [];
    
    for (var r = 1; r < data.length; r++) {
      var rolFila = (data[r][colRol] || "").toString().toUpperCase().trim();
      if (rolFila === rolUpper || rolFila === "ADMINISTRADOR") {
        var c = (data[r][colCorreo] || "").toString().toLowerCase().trim();
        if (c && correos.indexOf(c) === -1) correos.push(c);
      }
    }
    return correos;
  } catch (e) {
    Logger.log("Error en obtenerCorreosPorRol: " + e.toString());
    return [];
  }
}

// 8. MOTOR DE ENRUTAMIENTO AUTORIZANTE (MATRIZ DE 16 CASOS)
// --------------------------------------------------------------------------
function calcularRutaAutorizante(datosSolicitante, monto, dentroPresupuesto) {
  var idDbMaestro = obtenerParametro("ID_DB_MAESTRO");
  if (!idDbMaestro) throw new Error("ID_DB_MAESTRO no configurado en Parametros_Viaticos2.0.");
  var ssMaestro = SpreadsheetApp.openById(idDbMaestro);
  var sheetGerencias = ssMaestro.getSheetByName("DimGerencias");
  var sheetRegiones = ssMaestro.getSheetByName("DimRegiones");
  
  var dataGerencias = sheetGerencias.getDataRange().getValues();
  var headersGerencias = dataGerencias[0];
  var colGerNom = -1, colGerAut = -1, colGerCor = -1, colGerBak = -1, colGerHab = -1, colGerPla = -1, colGerJef = -1, colGerCas = -1;
  for (var i = 0; i < headersGerencias.length; i++) {
    var hg = headersGerencias[i].toString().trim();
    if (hg === "NombreGerencia") colGerNom = i;
    if (hg === "CorreoAutorizador") colGerAut = i;
    if (hg === "CorreoGerente") colGerCor = i;
    if (hg === "CorreoGerenteBackUp") colGerBak = i;
    if (hg === "HabilitarBackUp") colGerHab = i;
    if (hg === "Plaza") colGerPla = i;
    if (hg === "PlazaJefe") colGerJef = i;
    if (hg === "Caso") colGerCas = i;
  }

  // Función interna para obtener el correo autorizador efectivo de una gerencia
  function obtenerCorreoEfectivo(nombreBuscado) {
    var busc = (nombreBuscado || "").toString().toUpperCase().trim();
    for (var r = 1; r < dataGerencias.length; r++) {
      var nomFila = (dataGerencias[r][colGerNom] || "").toString().toUpperCase().trim();
      if (nomFila === busc || (busc && nomFila.indexOf(busc) > -1)) {
        var cAut = (dataGerencias[r][colGerAut] || "").toString().toLowerCase().trim();
        if (cAut) return cAut;
        var habBak = (dataGerencias[r][colGerHab] || "").toString().toUpperCase().trim();
        var cBak = (dataGerencias[r][colGerBak] || "").toString().toLowerCase().trim();
        var cTit = (dataGerencias[r][colGerCor] || "").toString().toLowerCase().trim();
        return (habBak === "SI" && cBak) ? cBak : cTit;
      }
    }
    return "";
  }

  var correoDirectorEjecutivo = obtenerCorreoEfectivo("DIRECCION EJECUTIVA") || "cesar.barahona@bancointegral.com.sv";
  var correoFinanzas = obtenerCorreoEfectivo("GERENCIA DE FINANZAS") || "imelda.zaldivar@bancointegral.com.sv";
  
  var gerenciaSolicitante = (datosSolicitante.Gerencia || datosSolicitante.NombreGerencia || "").toString().toUpperCase().trim();
  var plazaSolicitante = (datosSolicitante.Cargo || datosSolicitante.Plaza || "").toString().toUpperCase().trim();
  var correoSolicitante = (datosSolicitante.Correo || datosSolicitante.CorreoUsuario || "").toString().toLowerCase().trim();
  
  var correoGerenteSolicitante = obtenerCorreoEfectivo(gerenciaSolicitante);
  var casoPlaza = 1; // Por defecto colaborador regular (Caso 1)
  var plazaJefeDirecto = "";
  var correoJefeDirecto = "";

  // Verificar si el solicitante es Titular de Gerencia (Caso 2 o 3) o Director Ejecutivo (Caso 4)
  for (var g = 1; g < dataGerencias.length; g++) {
    var cTitular = (dataGerencias[g][colGerCor] || "").toString().toLowerCase().trim();
    var pTitular = (dataGerencias[g][colGerPla] || "").toString().toUpperCase().trim();
    var pJefe = (dataGerencias[g][colGerJef] || "").toString().toUpperCase().trim();
    var casoNum = parseInt(dataGerencias[g][colGerCas] || 1);

    if (cTitular === correoSolicitante || (plazaSolicitante && plazaSolicitante === pTitular)) {
      plazaJefeDirecto = pJefe;
      casoPlaza = casoNum;
      break;
    }
  }

  if (plazaJefeDirecto) {
    for (var j = 1; j < dataGerencias.length; j++) {
      if ((dataGerencias[j][colGerPla] || "").toString().toUpperCase().trim() === plazaJefeDirecto.toUpperCase()) {
        correoJefeDirecto = (dataGerencias[j][colGerAut] || dataGerencias[j][colGerCor] || "").toString().toLowerCase().trim();
        break;
      }
    }
  }
  if (!correoJefeDirecto) correoJefeDirecto = correoDirectorEjecutivo;
  if (!correoGerenteSolicitante) correoGerenteSolicitante = correoDirectorEjecutivo;

  if (plazaSolicitante.indexOf("DIRECTOR EJECUTIVO") > -1 || correoSolicitante === correoDirectorEjecutivo.toLowerCase()) {
    casoPlaza = 4;
  }

  var esAgencia = (gerenciaSolicitante.indexOf("AGENCIAS") > -1);
  var correoJefeRegional = "";

  if (esAgencia) {
    var dataRegiones = sheetRegiones.getDataRange().getValues();
    var headersReg = dataRegiones[0];
    var colRegNom = headersReg.indexOf("NombreRegion") !== -1 ? headersReg.indexOf("NombreRegion") : 1;
    var colRegJefe = headersReg.indexOf("CorreoJefeRegional") !== -1 ? headersReg.indexOf("CorreoJefeRegional") : 2;
    var regionSolicitante = (datosSolicitante.Region || datosSolicitante.NombreRegion || "").toString().toUpperCase().trim();
    for (var r = 1; r < dataRegiones.length; r++) {
      if ((dataRegiones[r][colRegNom] || "").toString().toUpperCase().trim() === regionSolicitante) {
        correoJefeRegional = (dataRegiones[r][colRegJefe] || "").toString().toLowerCase().trim();
        break;
      }
    }
    if (!correoJefeRegional) correoJefeRegional = correoGerenteSolicitante;
  }

  var ruta = {
    autorizacionesPendientes: 0,
    actorActual: "",
    estadoInicial: "INICIADO",
    correoJefeRegional: "",
    correoAutorizador1: "",
    correoAutorizador2: "",
    correoAutorizador3: ""
  };

  var montoNum = parseFloat(monto || 0);
  var esDentro = (dentroPresupuesto === "Si" || dentroPresupuesto === true);

  if (esDentro) {
    // --- RUTAS CON PRESUPUESTO ---
    if (casoPlaza === 4) {
      ruta.autorizacionesPendientes = 0;
      ruta.estadoInicial = "AUTORIZADO";
      ruta.actorActual = "Director Ejecutivo";
    } else if (casoPlaza === 3) {
      ruta.autorizacionesPendientes = 1;
      ruta.actorActual = "Autorizador 1";
      ruta.correoAutorizador1 = correoDirectorEjecutivo;
    } else if (casoPlaza === 2) {
      ruta.correoAutorizador1 = correoJefeDirecto;
      ruta.actorActual = "Autorizador 1";
      if (montoNum <= UMBRALES_MONTO.X1) {
        ruta.autorizacionesPendientes = 1;
      } else {
        ruta.autorizacionesPendientes = 2;
        ruta.correoAutorizador2 = correoDirectorEjecutivo;
      }
    } else { // Caso 1 (Colaborador Regular)
      if (!esAgencia) { // Back-Office
        ruta.actorActual = "Autorizador 1";
        ruta.correoAutorizador1 = correoGerenteSolicitante;
        if (montoNum <= UMBRALES_MONTO.X1) {
          ruta.autorizacionesPendientes = 1;
          // Solo nivel 1: autorizador 2 y 3 quedan vacíos
        } else if (montoNum <= UMBRALES_MONTO.X2) {
          ruta.autorizacionesPendientes = 2;
          ruta.correoAutorizador2 = correoFinanzas;
        } else {
          ruta.autorizacionesPendientes = 3;
          ruta.correoAutorizador2 = correoFinanzas;
          ruta.correoAutorizador3 = correoDirectorEjecutivo;
        }
      } else { // Agencias
        ruta.actorActual = "Autorizador 0";
        ruta.correoJefeRegional = correoJefeRegional;
        ruta.correoAutorizador1 = correoGerenteSolicitante;
        if (montoNum <= UMBRALES_MONTO.X1) {
          ruta.autorizacionesPendientes = 2;
        } else if (montoNum <= UMBRALES_MONTO.X2) {
          ruta.autorizacionesPendientes = 3;
          ruta.correoAutorizador2 = correoFinanzas;
        } else {
          ruta.autorizacionesPendientes = 4;
          ruta.correoAutorizador2 = correoFinanzas;
          ruta.correoAutorizador3 = correoDirectorEjecutivo;
        }
      }
    }
  } else {
    // --- RUTAS SIN PRESUPUESTO ---
    if (casoPlaza === 4) {
      ruta.autorizacionesPendientes = 0;
      ruta.estadoInicial = "AUTORIZADO";
      ruta.actorActual = "Director Ejecutivo";
    } else if (casoPlaza === 3 || casoPlaza === 2) {
      ruta.autorizacionesPendientes = 1;
      ruta.actorActual = "Autorizador 1";
      ruta.correoAutorizador1 = correoDirectorEjecutivo;
    } else { // Caso 1
      if (!esAgencia) {
        ruta.autorizacionesPendientes = 2;
        ruta.actorActual = "Autorizador 1";
        ruta.correoAutorizador1 = correoGerenteSolicitante;
        ruta.correoAutorizador2 = correoDirectorEjecutivo;
      } else {
        ruta.autorizacionesPendientes = 3;
        ruta.actorActual = "Autorizador 0";
        ruta.correoJefeRegional = correoJefeRegional;
        ruta.correoAutorizador1 = correoGerenteSolicitante;
        ruta.correoAutorizador2 = correoDirectorEjecutivo;
      }
    }
  }

  return ruta;
}

/**
 * Envía la tarjeta HTML de autorización al autorizador en turno utilizando GmailApp/MailApp,
 * asignando el nombre de remitente "Notificaciones_Viaticos2.0".
 */
function enviarTarjetaEmailAutorizador(correoAutorizador, idSolicitud, datosSolicitud, nivel) {
  try {
    if (!correoAutorizador) {
      Logger.log("Aviso: No se proporcionó correo de autorizador para solicitud " + idSolicitud);
      return { status: 'ERROR', mensaje: 'Correo de autorizador no especificado.' };
    }

    var webAppUrl = obtenerUrlWebApp();
    var nivelNum = nivel || 1;
    var htmlCuerpo = generarHtmlCardEmail(datosSolicitud, idSolicitud, nivelNum, webAppUrl);
    var nombreSolicitante = datosSolicitud.nombre_solicitante || datosSolicitud.NombreSolicitante || "Colaborador";
    var asunto = "Solicitud de Viáticos Pendiente de Autorización: " + idSolicitud + " - " + nombreSolicitante;
    var remitenteNombre = "Notificaciones_Viaticos2.0";

    Logger.log("Enviando correo HTML de autorización a: " + correoAutorizador + " (" + idSolicitud + ")");

    try {
      GmailApp.sendEmail(correoAutorizador, asunto, "", {
        htmlBody: htmlCuerpo,
        name: remitenteNombre
      });
      return { status: 'OK', enviado: true, canal: 'GmailApp', destinatario: correoAutorizador };
    } catch (eGmail) {
      Logger.log("Intento con MailApp por excepción en GmailApp: " + eGmail.message);
      MailApp.sendEmail({
        to: correoAutorizador,
        subject: asunto,
        htmlBody: htmlCuerpo,
        name: remitenteNombre
      });
      return { status: 'OK', enviado: true, canal: 'MailApp', destinatario: correoAutorizador };
    }
  } catch (error) {
    Logger.log("Error crítico al enviar tarjeta por correo: " + error.toString());
    return { status: 'ERROR', mensaje: error.toString() };
  }
}

/**
 * Consulta la información completa de una solicitud para alimentar la vista de resolución web (doGet).
 */
function obtenerDetalleSolicitudParaResolucion(idSolicitud) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    if (data.length < 2) return { status: 'ERROR', mensaje: 'La base transaccional está vacía.' };
    
    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;
    
    var filaEncontrada = -1;
    for (var r = 1; r < data.length; r++) {
      if ((data[r][colMap["ID_Solicitud"]] || "").toString().trim() === idSolicitud.toString().trim()) {
        filaEncontrada = r;
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró el expediente ' + idSolicitud + ' en la base de datos.' };
    }
    
    var row = data[filaEncontrada];
    var estado = (row[colMap["EstadoSolicitud"]] || "").toString().trim();
    var pendientes = parseInt(row[colMap["AutorizacionesPendientes"]] || 0);
    var yaResuelto = (estado === "AUTORIZADO" || estado === "RECHAZADO POR AUTORIZADOR" || pendientes <= 0);

    var bitacora = [];
    try {
      var rawBit = row[colMap["DatosAutorizacion"]];
      if (rawBit) bitacora = JSON.parse(rawBit);
    } catch (e) { bitacora = []; }

    return {
      status: 'OK',
      idSolicitud: idSolicitud,
      nombreSolicitante: row[colMap["NombreSolicitante"]] || "",
      correoSolicitante: row[colMap["CorreoSolicitante"]] || "",
      cargoSolicitante: row[colMap["CargoSolicitante"]] || "",
      gerencia: row[colMap["Gerencia"]] || "",
      centroCosto: row[colMap["CentroCosto"]] || "",
      agencia: row[colMap["Agencia"]] || "",
      duracionActividad: row[colMap["DuracionActividad"]] || "Día único",
      fechaInicio: row[colMap["FechaInicio"]] || "",
      fechaFin: row[colMap["FechaFin"]] || "",
      tipoViatico: row[colMap["TipoViatico"]] || "",
      horaEvento: row[colMap["HoraEvento"]] || "",
      monto: parseFloat(row[colMap["Monto"]] || 0),
      motivoViatico: row[colMap["MotivoViatico"]] || "",
      clasificacionSolicitud: row[colMap["ClasificacionSolicitud"]] || "Anticipo",
      dentroPresupuesto: row[colMap["DentroPresupuesto"]] || "Si",
      estadoSolicitud: estado,
      autorizacionesPendientes: pendientes,
      actorActual: row[colMap["ActorActual"]] || "",
      yaResuelto: yaResuelto,
      datosAutorizacion: bitacora
    };
  } catch (err) {
    return { status: 'ERROR', mensaje: 'Error al consultar la solicitud: ' + err.toString() };
  }
}

/**
 * Procesa la confirmación definitiva de resolución (Aprobación o Rechazo) emitida desde la página web.
 */
function procesarResolucionWeb(idSolicitud, resolucion, nivel, comentario, correoAutorizador, nombreAutorizador) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;
    
    var filaEncontrada = -1;
    for (var r = 1; r < data.length; r++) {
      if ((data[r][colMap["ID_Solicitud"]] || "").toString().trim() === idSolicitud.toString().trim()) {
        filaEncontrada = r + 1; // 1-indexed
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la solicitud ' + idSolicitud };
    }
    
    var filaIndex = filaEncontrada - 1;
    var estadoActual = (data[filaIndex][colMap["EstadoSolicitud"]] || "").toString().trim();
    var pendientesActual = parseInt(data[filaIndex][colMap["AutorizacionesPendientes"]] || 0);
    
    // Verificación de concurrencia y bloqueo de resolución previa
    if (estadoActual === "AUTORIZADO" || estadoActual === "RECHAZADO POR AUTORIZADOR" || pendientesActual <= 0) {
      return {
        status: 'ERROR',
        mensaje: 'Esta solicitud ya fue resuelta previamente.',
        yaResuelto: true,
        estadoActual: estadoActual
      };
    }
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var emailAutorizadorFinal = correoAutorizador || Session.getActiveUser().getEmail() || "";
    var nombreAutorizadorFinal = nombreAutorizador || emailAutorizadorFinal.split('@')[0] || "Autorizador Nivel " + nivel;

    // Actualizar Bitácora JSON
    var bitacoraExistente = [];
    try {
      var rawBitacora = data[filaIndex][colMap["DatosAutorizacion"]];
      if (rawBitacora) bitacoraExistente = JSON.parse(rawBitacora);
    } catch (e) { bitacoraExistente = []; }
    
    bitacoraExistente.push({
      nivel: nivel || 1,
      nombre: nombreAutorizadorFinal,
      correo: emailAutorizadorFinal,
      comentario: (comentario || "").trim() || (resolucion === "APROBADO" ? "Aprobado sin observaciones." : "Rechazado."),
      resolucion: resolucion,
      fecha: fechaHoyFormato
    });
    
    var nuevoEstado = estadoActual;
    var nuevosPendientes = pendientesActual;
    var nuevoActor = data[filaIndex][colMap["ActorActual"]];
    
    if (resolucion === "RECHAZADO") {
      nuevoEstado = "RECHAZADO POR AUTORIZADOR";
      nuevosPendientes = 0;
      nuevoActor = "Rechazado";
    } else if (resolucion === "APROBADO") {
      nuevosPendientes = pendientesActual - 1;
      if (nuevosPendientes <= 0) {
        nuevoEstado = "AUTORIZADO";
        nuevoActor = "Compras";
      } else {
        nuevoEstado = "INICIADO";
        // Escalar al siguiente actor autorizante mediante correo
        if (nuevoActor === "Autorizador 0") {
          nuevoActor = "Autorizador 1";
          var cAut1 = data[filaIndex][colMap["CorreoAutorizador1"]];
          if (cAut1) enviarTarjetaEmailAutorizador(cAut1, idSolicitud, {
            nombre_solicitante: data[filaIndex][colMap["NombreSolicitante"]],
            cargo_solicitante: data[filaIndex][colMap["CargoSolicitante"]],
            correo_solicitante: data[filaIndex][colMap["CorreoSolicitante"]],
            gerencia_solicitante: data[filaIndex][colMap["Gerencia"]],
            centro_costo: data[filaIndex][colMap["CentroCosto"]],
            agencia: data[filaIndex][colMap["Agencia"]],
            monto: data[filaIndex][colMap["Monto"]],
            tipo_viatico: data[filaIndex][colMap["TipoViatico"]],
            duracion_actividad: data[filaIndex][colMap["DuracionActividad"]],
            fecha_inicio: data[filaIndex][colMap["FechaInicio"]],
            fecha_fin: data[filaIndex][colMap["FechaFin"]],
            motivo_viatico: data[filaIndex][colMap["MotivoViatico"]],
            clasificacion_solicitud: data[filaIndex][colMap["ClasificacionSolicitud"]],
            dentroPresupuesto: data[filaIndex][colMap["DentroPresupuesto"]]
          }, 1);
        } else if (nuevoActor === "Autorizador 1") {
          nuevoActor = "Autorizador 2";
          var cAut2 = data[filaIndex][colMap["CorreoAutorizador2"]];
          if (cAut2) enviarTarjetaEmailAutorizador(cAut2, idSolicitud, {
            nombre_solicitante: data[filaIndex][colMap["NombreSolicitante"]],
            cargo_solicitante: data[filaIndex][colMap["CargoSolicitante"]],
            correo_solicitante: data[filaIndex][colMap["CorreoSolicitante"]],
            gerencia_solicitante: data[filaIndex][colMap["Gerencia"]],
            centro_costo: data[filaIndex][colMap["CentroCosto"]],
            agencia: data[filaIndex][colMap["Agencia"]],
            monto: data[filaIndex][colMap["Monto"]],
            tipo_viatico: data[filaIndex][colMap["TipoViatico"]],
            duracion_actividad: data[filaIndex][colMap["DuracionActividad"]],
            fecha_inicio: data[filaIndex][colMap["FechaInicio"]],
            fecha_fin: data[filaIndex][colMap["FechaFin"]],
            motivo_viatico: data[filaIndex][colMap["MotivoViatico"]],
            clasificacion_solicitud: data[filaIndex][colMap["ClasificacionSolicitud"]],
            dentroPresupuesto: data[filaIndex][colMap["DentroPresupuesto"]]
          }, 2);
        } else if (nuevoActor === "Autorizador 2") {
          nuevoActor = "Autorizador 3";
          var cAut3 = data[filaIndex][colMap["CorreoAutorizador3"]];
          if (cAut3) enviarTarjetaEmailAutorizador(cAut3, idSolicitud, {
            nombre_solicitante: data[filaIndex][colMap["NombreSolicitante"]],
            cargo_solicitante: data[filaIndex][colMap["CargoSolicitante"]],
            correo_solicitante: data[filaIndex][colMap["CorreoSolicitante"]],
            gerencia_solicitante: data[filaIndex][colMap["Gerencia"]],
            centro_costo: data[filaIndex][colMap["CentroCosto"]],
            agencia: data[filaIndex][colMap["Agencia"]],
            monto: data[filaIndex][colMap["Monto"]],
            tipo_viatico: data[filaIndex][colMap["TipoViatico"]],
            duracion_actividad: data[filaIndex][colMap["DuracionActividad"]],
            fecha_inicio: data[filaIndex][colMap["FechaInicio"]],
            fecha_fin: data[filaIndex][colMap["FechaFin"]],
            motivo_viatico: data[filaIndex][colMap["MotivoViatico"]],
            clasificacion_solicitud: data[filaIndex][colMap["ClasificacionSolicitud"]],
            dentroPresupuesto: data[filaIndex][colMap["DentroPresupuesto"]]
          }, 3);
        }
      }
    }
    
    // Escribir actualizaciones en la hoja
    sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue(nuevoEstado);
    sheetFact.getRange(filaEncontrada, colMap["AutorizacionesPendientes"] + 1).setValue(nuevosPendientes);
    sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue(nuevoActor);
    sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
    sheetFact.getRange(filaEncontrada, colMap["DatosAutorizacion"] + 1).setValue(JSON.stringify(bitacoraExistente));
    
    return {
      status: 'OK',
      idSolicitud: idSolicitud,
      resolucion: resolucion,
      nuevoEstado: nuevoEstado,
      nuevosPendientes: nuevosPendientes,
      mensaje: 'Resolución registrada exitosamente. Estado actual: ' + nuevoEstado
    };
  } catch (error) {
    Logger.log("Error al procesar resolución web: " + error.toString());
    return { status: 'ERROR', mensaje: error.toString() };
  }
}

/**
 * Consulta la bandeja de solicitudes pendientes de autorización asignadas al usuario activo (A1).
 * Filtra solicitudes en estado 'INICIADO' o 'RECHAZO-PRESUPUESTO' donde ActorActual coincide con el autorizador.
 */
function obtenerSolicitudesAutorizacionA1(correoUsuario) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONAL") || obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return crearRespuestaAPI(false, null, 'No se encontró ID_DB_TRANSACCIONAL en Parametros_Viaticos2.0.');
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    if (data.length < 2) return crearRespuestaAPI(true, [], 'No hay solicitudes registradas.');
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var correoBuscado = (correoUsuario || Session.getActiveUser().getEmail() || "").toString().toLowerCase().trim();
    var solicitudes = [];
    
    for (var r = 1; r < data.length; r++) {
      var row = data[r];
      var estado = obtenerColVal(row, colMap, ["EstadoSolicitud", "Estado"], "").toString().trim();
      var actor = obtenerColVal(row, colMap, ["ActorActual", "Actor_Actual"], "").toString().trim();
      var pendientes = parseInt(obtenerColVal(row, colMap, ["AutorizacionesPendientes", "Pendientes"], 0)) || 0;
      
      // Solo solicitudes en curso de autorización
      if ((estado === "INICIADO" || estado === "RECHAZO-PRESUPUESTO") && pendientes > 0 && actor) {
        var esAsignado = false;
        
        var cJefeReg = obtenerColVal(row, colMap, ["CorreoJefeRegional"], "").toString().toLowerCase().trim();
        var cAut1 = obtenerColVal(row, colMap, ["CorreoAutorizador1"], "").toString().toLowerCase().trim();
        var cAut2 = obtenerColVal(row, colMap, ["CorreoAutorizador2"], "").toString().toLowerCase().trim();
        var cAut3 = obtenerColVal(row, colMap, ["CorreoAutorizador3"], "").toString().toLowerCase().trim();
        
        if (actor === "Autorizador 0" && cJefeReg === correoBuscado) {
          esAsignado = true;
        } else if (actor === "Autorizador 1" && cAut1 === correoBuscado) {
          esAsignado = true;
        } else if (actor === "Autorizador 2" && cAut2 === correoBuscado) {
          esAsignado = true;
        } else if (actor === "Autorizador 3" && cAut3 === correoBuscado) {
          esAsignado = true;
        } else if (!correoBuscado) {
          esAsignado = true;
        } else if (correoBuscado === cJefeReg || correoBuscado === cAut1 || correoBuscado === cAut2 || correoBuscado === cAut3) {
          if (actor.indexOf("Autorizador") > -1) {
            esAsignado = true;
          }
        }
        
        if (esAsignado) {
          var montoVal = parseFloat(obtenerColVal(row, colMap, ["Monto", "MontoTotal"], 0)) || 0;
          solicitudes.push({
            idSolicitud: obtenerColVal(row, colMap, ["ID_Solicitud"], "").toString().trim(),
            fechaSolicitud: obtenerColVal(row, colMap, ["FechaSolicitud", "Fecha_Solicitud"], "").toString().trim(),
            fechaModificacion: obtenerColVal(row, colMap, ["FechaModificacion", "Fecha_Modificacion"], "").toString().trim(),
            nombreSolicitante: obtenerColVal(row, colMap, ["NombreSolicitante", "Solicitante"], "").toString().trim(),
            correoSolicitante: obtenerColVal(row, colMap, ["CorreoSolicitante", "Correo"], "").toString().trim(),
            cargoSolicitante: obtenerColVal(row, colMap, ["CargoSolicitante", "Plaza"], "").toString().trim(),
            gerencia: obtenerColVal(row, colMap, ["Gerencia", "NombreGerencia"], "").toString().trim(),
            centroCosto: obtenerColVal(row, colMap, ["CentroCosto", "Centro_Costo"], "").toString().trim(),
            agencia: obtenerColVal(row, colMap, ["Agencia", "NombreAgencia"], "Oficina Central").toString().trim(),
            duracionActividad: obtenerColVal(row, colMap, ["DuracionActividad", "Duracion"], "Día único").toString().trim(),
            fechaInicio: obtenerColVal(row, colMap, ["FechaInicio", "Fecha_Inicio"], "").toString().trim(),
            fechaFin: obtenerColVal(row, colMap, ["FechaFin", "Fecha_Fin"], "").toString().trim(),
            tipoViatico: obtenerColVal(row, colMap, ["TipoViatico", "Tipo_Viatico"], "").toString().trim(),
            destinatario: obtenerColVal(row, colMap, ["Destinatario", "NombreSolicitante"], "").toString().trim(),
            monto: montoVal,
            montoFormatted: "$ " + montoVal.toFixed(2),
            estadoSolicitud: estado,
            actorActual: actor,
            clasificacionSolicitud: obtenerColVal(row, colMap, ["ClasificacionSolicitud", "Clasificacion"], "Anticipo").toString().trim(),
            dentroPresupuesto: obtenerColVal(row, colMap, ["DentroPresupuesto", "Dentro_Presupuesto"], "Si").toString().trim(),
            autorizacionesPendientes: pendientes
          });
        }
      }
    }
    
    var resp = crearRespuestaAPI(true, solicitudes, 'Autorizaciones recuperadas.');
    resp.solicitudes = solicitudes;
    return resp;
  } catch (err) {
    Logger.log("Error en obtenerSolicitudesAutorizacionA1: " + err.toString());
    return crearRespuestaAPI(false, null, 'Error al consultar autorizaciones: ' + err.toString());
  }
}

/**
 * Retorna la información completa de la solicitud para la pantalla de resolución confidencial (A1.1).
 * Incluye todos los campos de solicitud, detalle bancario completo y galería de adjuntos.
 */
function obtenerDetalleAutorizacionA1_1(idSolicitud) {
  try {
    if (!idSolicitud) return crearRespuestaAPI(false, null, 'ID de solicitud no proporcionado.');
    var idBuscado = idSolicitud.toString().trim();
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONAL") || 
                            obtenerParametro("ID_DB_TRANSACCIONES") || 
                            obtenerParametro("ID_TRANSACCIONES") || 
                            obtenerParametro("ID_BASE_DATOS_VIATICOS");
    if (!idDbTransaccional) return crearRespuestaAPI(false, null, 'No se encontró ID_DB_TRANSACCIONAL en Parametros.');
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheetByName("Transaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    if (data.length < 2) return crearRespuestaAPI(false, null, 'Base de datos transaccional vacía.');
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    var colId = obtenerIndiceColumna(colMap, "ID_Solicitud", 0);
    
    var filaEncontrada = -1;
    var idBuscadoUpper = idBuscado.toUpperCase();
    
    for (var r = 1; r < data.length; r++) {
      var celdaId = (data[r][colId] || "").toString().trim();
      if (celdaId.toUpperCase() === idBuscadoUpper) {
        filaEncontrada = r;
        break;
      }
    }
    
    // Fallback: búsqueda lineal por todas las celdas de la fila si colId fue desplazada
    if (filaEncontrada === -1) {
      for (var r = 1; r < data.length; r++) {
        for (var c = 0; c < Math.min(data[r].length, 5); c++) {
          if ((data[r][c] || "").toString().trim().toUpperCase() === idBuscadoUpper) {
            filaEncontrada = r;
            break;
          }
        }
        if (filaEncontrada !== -1) break;
      }
    }
    
    if (filaEncontrada === -1) {
      return crearRespuestaAPI(false, null, 'No se encontró el expediente ' + idSolicitud);
    }
    
    var row = data[filaEncontrada];
    var estado = obtenerColVal(row, colMap, ["EstadoSolicitud", "Estado", "EstadoSolicitudViatico"], "INICIADO").toString().trim();
    var pendientes = parseInt(obtenerColVal(row, colMap, ["AutorizacionesPendientes", "Pendientes", "NivelesPendientes"], 0)) || 0;
    var yaResuelto = (estado === "AUTORIZADO" || estado === "RECHAZO-AUTORIZACION");
    
    var bitacora = [];
    try {
      var rawBit = obtenerColVal(row, colMap, ["DatosAutorizacion", "BitacoraAutorizacion", "HistorialDeFirmas"], "");
      if (rawBit) {
        bitacora = (typeof rawBit === 'string') ? JSON.parse(rawBit) : rawBit;
        if (!Array.isArray(bitacora)) bitacora = [bitacora];
      }
    } catch (e) { bitacora = []; }
    
    var archivos = [];
    try {
      var rawArch = obtenerColVal(row, colMap, ["ArchivosAdjuntos", "Archivos", "Adjuntos"], "");
      if (rawArch) {
        archivos = (typeof rawArch === 'string') ? JSON.parse(rawArch) : rawArch;
        if (!Array.isArray(archivos)) archivos = [archivos];
      }
    } catch (e) { archivos = []; }
    
    var montoVal = parseFloat(obtenerColVal(row, colMap, ["Monto", "MontoTotal", "Total"], 0)) || 0;
    
    var detalle = {
      idSolicitud: idBuscado,
      nombreSolicitante: obtenerColVal(row, colMap, ["NombreSolicitante", "Solicitante", "Nombre"], "").toString().trim(),
      correoSolicitante: obtenerColVal(row, colMap, ["CorreoSolicitante", "Correo", "Email"], "").toString().trim(),
      cargoSolicitante: obtenerColVal(row, colMap, ["CargoSolicitante", "Plaza", "Cargo"], "").toString().trim(),
      gerencia: obtenerColVal(row, colMap, ["Gerencia", "NombreGerencia", "Departamento"], "").toString().trim(),
      centroCosto: obtenerColVal(row, colMap, ["CentroCosto", "Centro_Costo", "CentroDeCosto", "CECO"], "").toString().trim(),
      agencia: obtenerColVal(row, colMap, ["Agencia", "NombreAgencia", "Sucursal"], "Oficina Central").toString().trim(),
      duracionActividad: obtenerColVal(row, colMap, ["DuracionActividad", "Duracion", "DuracionDeLaActividad"], "Día único").toString().trim(),
      fechaInicio: obtenerColVal(row, colMap, ["FechaInicio", "Fecha_Inicio", "Desde"], "").toString().trim(),
      fechaFin: obtenerColVal(row, colMap, ["FechaFin", "Fecha_Fin", "Hasta"], "").toString().trim(),
      tipoViatico: obtenerColVal(row, colMap, ["TipoViatico", "Tipo_Viatico", "RubroViatico", "TipoDeViatico"], "").toString().trim(),
      horaEvento: obtenerColVal(row, colMap, ["HoraEvento", "Hora_Evento", "HoraDelEvento"], "").toString().trim(),
      tipoSolicitud: obtenerColVal(row, colMap, ["TipoSolicitud", "Tipo_Solicitud", "TipoDeSolicitud"], "Personal").toString().trim(),
      destinatario: obtenerColVal(row, colMap, ["Destinatario", "NombreDestinatario", "NombreSolicitante"], "").toString().trim(),
      correoDestinatario: obtenerColVal(row, colMap, ["CorreoDestinatario", "CorreoSolicitante"], "").toString().trim(),
      monto: montoVal,
      montoFormatted: "$ " + montoVal.toFixed(2),
      banco: obtenerColVal(row, colMap, ["Banco", "NombreBanco", "EntidadBancaria"], "Banco Integral").toString().trim(),
      tipoCuenta: obtenerColVal(row, colMap, ["TipoCuenta", "Tipo_Cuenta", "TipoDeCuenta"], "Ahorro").toString().trim(),
      numeroCuenta: obtenerColVal(row, colMap, ["NumeroCuenta", "Numero_Cuenta", "NumeroDeCuenta", "NoDeCuenta"], "-").toString().trim(),
      esEditado: obtenerColVal(row, colMap, ["EsEditado", "Es_Editado", "InformacionEditada"], "No").toString().trim(),
      motivoViatico: obtenerColVal(row, colMap, ["MotivoViatico", "Motivo", "MotivoDelViatico", "Justificacion"], "").toString().trim(),
      rubroContable: obtenerColVal(row, colMap, ["RubroContable", "Rubro", "CuentaContable"], "").toString().trim(),
      clasificacionSolicitud: obtenerColVal(row, colMap, ["ClasificacionSolicitud", "Clasificacion", "ClasificacionDeSolicitud"], "Anticipo").toString().trim(),
      dentroPresupuesto: obtenerColVal(row, colMap, ["DentroPresupuesto", "Dentro_Presupuesto", "DentroDePresupuesto", "EstadoDelPresupuesto"], "Si").toString().trim(),
      justificacionPresupuesto: obtenerColVal(row, colMap, ["JustificacionPresupuesto", "JustifcacionPresupuesto", "Justificacion_Presupuesto", "JustificacionFueraDePresupuesto", "JustificacionExtraordinaria"], "").toString().trim(),
      estadoSolicitud: estado,
      autorizacionesPendientes: pendientes,
      actorActual: obtenerColVal(row, colMap, ["ActorActual", "Actor_Actual", "Responsable"], "").toString().trim(),
      yaResuelto: yaResuelto,
      datosAutorizacion: bitacora,
      archivosAdjuntos: archivos
    };

    var res = crearRespuestaAPI(true, detalle, 'Expediente recuperado con éxito.');
    res.solicitud = detalle;
    return res;
  } catch (err) {
    Logger.log("Error en obtenerDetalleAutorizacionA1_1: " + err.toString());
    return crearRespuestaAPI(false, null, 'Error al consultar el expediente: ' + err.toString());
  }
}

/**
 * Procesa la resolución (APROBADO o RECHAZADO) emitida por un autorizador (A1.1).
 */
function procesarResolucionA1_1(idSolicitud, resolucion, comentario, usuario) {
  try {
    if (!idSolicitud) return crearRespuestaAPI(false, null, 'ID de solicitud requerido.');
    
    var resolucionUpper = (resolucion || "").toString().toUpperCase().trim();
    if (resolucionUpper !== "APROBAR" && resolucionUpper !== "APROBADO" && resolucionUpper !== "RECHAZAR" && resolucionUpper !== "RECHAZADO") {
      return crearRespuestaAPI(false, null, 'Resolución inválida. Debe ser Aprobado o Rechazado.');
    }
    
    var esRechazo = (resolucionUpper === "RECHAZAR" || resolucionUpper === "RECHAZADO");
    var comTrim = (comentario || "").toString().trim();
    if (esRechazo && comTrim.length < 5) {
      return crearRespuestaAPI(false, null, 'Debe ingresar un motivo u observación de rechazo (mínimo 5 caracteres).');
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONAL") || 
                            obtenerParametro("ID_DB_TRANSACCIONES") || 
                            obtenerParametro("ID_TRANSACCIONES") || 
                            obtenerParametro("ID_BASE_DATOS_VIATICOS");
    if (!idDbTransaccional) return crearRespuestaAPI(false, null, 'No se encontró ID_DB_TRANSACCIONAL en Parametros.');
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheetByName("Transaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    var colId = obtenerIndiceColumna(colMap, "ID_Solicitud", 0);
    
    var idBuscadoUpper = idSolicitud.toString().trim().toUpperCase();
    var filaEncontrada = -1;
    for (var r = 1; r < data.length; r++) {
      var celdaId = (data[r][colId] || "").toString().trim();
      if (celdaId.toUpperCase() === idBuscadoUpper) {
        filaEncontrada = r + 1; // 1-indexed para getRange
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      for (var r = 1; r < data.length; r++) {
        for (var c = 0; c < Math.min(data[r].length, 5); c++) {
          if ((data[r][c] || "").toString().trim().toUpperCase() === idBuscadoUpper) {
            filaEncontrada = r + 1;
            break;
          }
        }
        if (filaEncontrada !== -1) break;
      }
    }
    
    if (filaEncontrada === -1) {
      return crearRespuestaAPI(false, null, 'No se encontró la solicitud ' + idSolicitud);
    }
    
    var filaIndex = filaEncontrada - 1;
    var estadoActual = obtenerColVal(data[filaIndex], colMap, ["EstadoSolicitud", "Estado"], "INICIADO").toString().trim();
    var pendientesActual = parseInt(obtenerColVal(data[filaIndex], colMap, ["AutorizacionesPendientes", "Pendientes"], 0)) || 0;
    
    // Si ya está en estado terminal resuelto, bloquear
    if (estadoActual === "AUTORIZADO" || estadoActual === "RECHAZO-AUTORIZACION") {
      return crearRespuestaAPI(false, { yaResuelto: true, estadoActual: estadoActual }, 'Esta solicitud ya fue resuelta previamente.');
    }
    
    // Si pendientesActual es <= 0 pero el estado es INICIADO o RECHAZO-PRESUPUESTO, interpretar como 1 pendiente para permitir aprobar
    if (pendientesActual <= 0 && (estadoActual === "INICIADO" || estadoActual === "RECHAZO-PRESUPUESTO")) {
      pendientesActual = 1;
    }
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var correoAut = (usuario && (usuario.CorreoGoogle || usuario.CorreoUsuario || usuario.email)) || Session.getActiveUser().getEmail() || "";
    var nombreAut = (usuario && (usuario.NombreGoogle || usuario.NombreUsuario || usuario.nombre)) || (correoAut ? correoAut.split('@')[0] : "Autorizador");
    var actorActual = obtenerColVal(data[filaIndex], colMap, ["ActorActual", "Actor_Actual", "Responsable"], "").toString().trim();
    
    // Bitácora de autorización
    var bitacoraExistente = [];
    try {
      var rawBitacora = obtenerColVal(data[filaIndex], colMap, ["DatosAutorizacion", "BitacoraAutorizacion", "HistorialDeFirmas"], "");
      if (rawBitacora) {
        bitacoraExistente = (typeof rawBitacora === 'string') ? JSON.parse(rawBitacora) : rawBitacora;
        if (!Array.isArray(bitacoraExistente)) bitacoraExistente = [bitacoraExistente];
      }
    } catch (e) { bitacoraExistente = []; }
    
    bitacoraExistente.push({
      nivel: actorActual || "Autorizador",
      nombre: nombreAut,
      correo: correoAut,
      comentario: comTrim || (esRechazo ? "Rechazado por autorizador." : "Aprobado sin observaciones."),
      resolucion: esRechazo ? "Rechazado" : "Aprobado",
      fecha: fechaHoyFormato
    });
    
    var nuevoEstado = estadoActual;
    var nuevosPendientes = pendientesActual;
    var nuevoActor = actorActual;
    
    if (esRechazo) {
      nuevoEstado = "RECHAZO-AUTORIZACION";
      nuevosPendientes = 0;
      nuevoActor = "";
      
      // Liberar presupuesto si estaba reservado
      var dentroPresupuesto = obtenerColVal(data[filaIndex], colMap, ["DentroPresupuesto", "Dentro_Presupuesto"], "Si").toString().trim();
      if (dentroPresupuesto === "Si" || dentroPresupuesto === "SI" || dentroPresupuesto === true) {
        var centroCosto = obtenerColVal(data[filaIndex], colMap, ["CentroCosto", "Centro_Costo"], "").toString().trim();
        var tipoViatico = obtenerColVal(data[filaIndex], colMap, ["TipoViatico", "Tipo_Viatico"], "").toString().trim();
        var montoVal = parseFloat(obtenerColVal(data[filaIndex], colMap, ["Monto", "MontoTotal"], 0)) || 0;
        var fechaSol = obtenerColVal(data[filaIndex], colMap, ["FechaSolicitud", "Fecha_Solicitud"], "").toString().trim();
        
        try {
          var centroNum = parseInt(centroCosto.split(" - ")[0] || centroCosto) || 0;
          revertirAfectacionPresupuesto(centroNum, tipoViatico, fechaSol, montoVal);
          Logger.log("Presupuesto liberado para " + idSolicitud + " por rechazo de autorizador ($" + montoVal + ")");
        } catch (ePres) {
          Logger.log("Aviso al liberar presupuesto en rechazo A1.1: " + ePres.toString());
        }
      }
    } else {
      // APROBAR
      nuevosPendientes = Math.max(0, pendientesActual - 1);
      if (nuevosPendientes <= 0) {
        nuevoEstado = "AUTORIZADO";
        nuevoActor = "Compras";
      } else {
        nuevoEstado = "INICIADO";
        // Escalar al siguiente autorizador en la ruta
        if (actorActual === "Autorizador 0") {
          nuevoActor = "Autorizador 1";
          var cAut1 = obtenerColVal(data[filaIndex], colMap, ["CorreoAutorizador1"], "").toString().trim();
          if (cAut1) enviarTarjetaEmailAutorizador(cAut1, idSolicitud, {
            nombre_solicitante: obtenerColVal(data[filaIndex], colMap, ["NombreSolicitante", "Solicitante"], ""),
            cargo_solicitante: obtenerColVal(data[filaIndex], colMap, ["CargoSolicitante", "Plaza"], ""),
            correo_solicitante: obtenerColVal(data[filaIndex], colMap, ["CorreoSolicitante", "Correo"], ""),
            gerencia_solicitante: obtenerColVal(data[filaIndex], colMap, ["Gerencia", "NombreGerencia"], ""),
            centro_costo: obtenerColVal(data[filaIndex], colMap, ["CentroCosto"], ""),
            agencia: obtenerColVal(data[filaIndex], colMap, ["Agencia", "NombreAgencia"], ""),
            monto: obtenerColVal(data[filaIndex], colMap, ["Monto"], 0),
            tipo_viatico: obtenerColVal(data[filaIndex], colMap, ["TipoViatico"], ""),
            duracion_actividad: obtenerColVal(data[filaIndex], colMap, ["DuracionActividad"], ""),
            fecha_inicio: obtenerColVal(data[filaIndex], colMap, ["FechaInicio"], ""),
            fecha_fin: obtenerColVal(data[filaIndex], colMap, ["FechaFin"], ""),
            motivo_viatico: obtenerColVal(data[filaIndex], colMap, ["MotivoViatico"], ""),
            clasificacion_solicitud: obtenerColVal(data[filaIndex], colMap, ["ClasificacionSolicitud"], ""),
            dentroPresupuesto: obtenerColVal(data[filaIndex], colMap, ["DentroPresupuesto"], "")
          }, 1);
        } else if (actorActual === "Autorizador 1") {
          nuevoActor = "Autorizador 2";
          var cAut2 = obtenerColVal(data[filaIndex], colMap, ["CorreoAutorizador2"], "").toString().trim();
          if (cAut2) enviarTarjetaEmailAutorizador(cAut2, idSolicitud, {
            nombre_solicitante: obtenerColVal(data[filaIndex], colMap, ["NombreSolicitante", "Solicitante"], ""),
            cargo_solicitante: obtenerColVal(data[filaIndex], colMap, ["CargoSolicitante", "Plaza"], ""),
            correo_solicitante: obtenerColVal(data[filaIndex], colMap, ["CorreoSolicitante", "Correo"], ""),
            gerencia_solicitante: obtenerColVal(data[filaIndex], colMap, ["Gerencia", "NombreGerencia"], ""),
            centro_costo: obtenerColVal(data[filaIndex], colMap, ["CentroCosto"], ""),
            agencia: obtenerColVal(data[filaIndex], colMap, ["Agencia", "NombreAgencia"], ""),
            monto: obtenerColVal(data[filaIndex], colMap, ["Monto"], 0),
            tipo_viatico: obtenerColVal(data[filaIndex], colMap, ["TipoViatico"], ""),
            duracion_actividad: obtenerColVal(data[filaIndex], colMap, ["DuracionActividad"], ""),
            fecha_inicio: obtenerColVal(data[filaIndex], colMap, ["FechaInicio"], ""),
            fecha_fin: obtenerColVal(data[filaIndex], colMap, ["FechaFin"], ""),
            motivo_viatico: obtenerColVal(data[filaIndex], colMap, ["MotivoViatico"], ""),
            clasificacion_solicitud: obtenerColVal(data[filaIndex], colMap, ["ClasificacionSolicitud"], ""),
            dentroPresupuesto: obtenerColVal(data[filaIndex], colMap, ["DentroPresupuesto"], "")
          }, 2);
        } else if (actorActual === "Autorizador 2") {
          nuevoActor = "Autorizador 3";
          var cAut3 = obtenerColVal(data[filaIndex], colMap, ["CorreoAutorizador3"], "").toString().trim();
          if (cAut3) enviarTarjetaEmailAutorizador(cAut3, idSolicitud, {
            nombre_solicitante: obtenerColVal(data[filaIndex], colMap, ["NombreSolicitante", "Solicitante"], ""),
            cargo_solicitante: obtenerColVal(data[filaIndex], colMap, ["CargoSolicitante", "Plaza"], ""),
            correo_solicitante: obtenerColVal(data[filaIndex], colMap, ["CorreoSolicitante", "Correo"], ""),
            gerencia_solicitante: obtenerColVal(data[filaIndex], colMap, ["Gerencia", "NombreGerencia"], ""),
            centro_costo: obtenerColVal(data[filaIndex], colMap, ["CentroCosto"], ""),
            agencia: obtenerColVal(data[filaIndex], colMap, ["Agencia", "NombreAgencia"], ""),
            monto: obtenerColVal(data[filaIndex], colMap, ["Monto"], 0),
            tipo_viatico: obtenerColVal(data[filaIndex], colMap, ["TipoViatico"], ""),
            duracion_actividad: obtenerColVal(data[filaIndex], colMap, ["DuracionActividad"], ""),
            fecha_inicio: obtenerColVal(data[filaIndex], colMap, ["FechaInicio"], ""),
            fecha_fin: obtenerColVal(data[filaIndex], colMap, ["FechaFin"], ""),
            motivo_viatico: obtenerColVal(data[filaIndex], colMap, ["MotivoViatico"], ""),
            clasificacion_solicitud: obtenerColVal(data[filaIndex], colMap, ["ClasificacionSolicitud"], ""),
            dentroPresupuesto: obtenerColVal(data[filaIndex], colMap, ["DentroPresupuesto"], "")
          }, 3);
        }
      }
    }
    
    // Guardar cambios en DimTransaccional con fallbacks canónicos de columna
    establecerColVal(sheetFact, filaEncontrada, colMap, "EstadoSolicitud", nuevoEstado, 3);
    establecerColVal(sheetFact, filaEncontrada, colMap, "AutorizacionesPendientes", nuevosPendientes, 4);
    establecerColVal(sheetFact, filaEncontrada, colMap, "ActorActual", nuevoActor, 5);
    establecerColVal(sheetFact, filaEncontrada, colMap, "FechaModificacion", fechaHoyFormato, 7);
    establecerColVal(sheetFact, filaEncontrada, colMap, "DatosAutorizacion", JSON.stringify(bitacoraExistente), 35);
    
    SpreadsheetApp.flush();
    
    var msgExito = esRechazo 
      ? 'La solicitud ' + idSolicitud + ' ha sido rechazada exitosamente.' 
      : (nuevosPendientes <= 0 
          ? '¡Autorización completada! El expediente ' + idSolicitud + ' ha sido aprobado en todos sus niveles y pasa a compras.' 
          : 'Firma registrada con éxito. La solicitud ' + idSolicitud + ' fue escalada a ' + nuevoActor + '.');

    return crearRespuestaAPI(true, {
      idSolicitud: idSolicitud,
      resolucion: resolucionUpper,
      nuevoEstado: nuevoEstado,
      nuevosPendientes: nuevosPendientes,
      nuevoActor: nuevoActor
    }, msgExito);

  } catch (error) {
    Logger.log("Error al procesar resolución A1.1: " + error.toString());
    return crearRespuestaAPI(false, null, 'Error en el servidor al registrar resolución: ' + error.toString());
  }
}

// 10. GUARDADO PRINCIPAL DE LA SOLICITUD S1 (DIMTRANSACCIONAL)
// --------------------------------------------------------------------------
function guardarNuevaSolicitudS1(datosEmpaquetados) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    
    var dataExistente = sheetFact.getDataRange().getValues();
    var correlativoNum = dataExistente.length;
    var correlativo = "SOL-" + new Date().getFullYear() + "-" + (correlativoNum < 10 ? "000" : correlativoNum < 100 ? "00" : "0") + correlativoNum;
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    
    // 1. Guardar archivos en Drive y serializar JSON
    var archivosEntrada = datosEmpaquetados.archivos || datosEmpaquetados.listaArchivos || datosEmpaquetados.archivosAdjuntos || [];
    var listaArchivosGuardados = guardarArchivosEnDrive(correlativo, archivosEntrada);
    var archivosJSON = JSON.stringify(listaArchivosGuardados);
    
    // 2. Validación de Presupuesto (El mes se extrae de FechaSolicitud)
    var centroCostoNum = datosEmpaquetados.centro_costo_num || (datosEmpaquetados.centro_costo || "").split('-')[0] || "1";
    var rubroCodigo = datosEmpaquetados.rubro_codigo || "8110050400000";
    var montoNum = parseFloat(datosEmpaquetados.monto || datosEmpaquetados.monto_total || 0);
    
    var resPresupuesto = validarYAfectarPresupuesto(centroCostoNum, rubroCodigo, fechaHoyFormato, montoNum);
    var dentroPresupuesto = (resPresupuesto.dentroPresupuesto || "Si");
    
    // 3. Cálculo de la Ruta de Autorización (16 Casos)
    var ruta = calcularRutaAutorizante({
      Gerencia: datosEmpaquetados.gerencia_solicitante,
      Plaza: datosEmpaquetados.cargo_solicitante,
      Correo: datosEmpaquetados.correo_solicitante,
      Region: datosEmpaquetados.region_solicitante
    }, montoNum, dentroPresupuesto);
    
    // 4. Cálculo de ClasificacionSolicitud ("Reintegro" vs "Anticipo")
    // Regla: Si FechaSolicitud - FechaInicio <= 2 días => "Reintegro", en otro caso "Anticipo"
    var clasificacionSolicitud = "Anticipo";
    try {
      var hoy = new Date();
      var fechaIniParts = (datosEmpaquetados.fecha_inicio || "").split('-');
      var fechaIni = fechaIniParts.length === 3 
        ? new Date(parseInt(fechaIniParts[0]), parseInt(fechaIniParts[1]) - 1, parseInt(fechaIniParts[2])) 
        : new Date(datosEmpaquetados.fecha_inicio);
      
      if (!isNaN(fechaIni.getTime())) {
        var diffDias = Math.floor((fechaIni.getTime() - hoy.getTime()) / (1000 * 60 * 60 * 24));
        if (diffDias <= 2) {
          clasificacionSolicitud = "Reintegro";
        } else {
          clasificacionSolicitud = "Anticipo";
        }
      }
    } catch (eClasif) {
      clasificacionSolicitud = "Anticipo";
    }

/**
 * Consulta sincrónica al cargar S1 para verificar si el usuario tiene solicitudes
 * pendientes de liquidar (PAGADO para anticipos o RECHAZO-CIERRE) con más de 15 días.
 * FechaReferencia = max(FechaFin o FechaInicio, FechaModificacion o FechaSolicitud).
 */
function verificarBloqueoSolicitudS1(correoUsuario) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    if (data.length < 2) return { status: 'OK', bloqueado: false, solicitudesBloqueantes: [] };
    
    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;
    
    var correoFiltro = (correoUsuario || "").toString().toLowerCase().trim();
    var hoy = new Date();
    hoy.setHours(0, 0, 0, 0);
    
    var solicitudesBloqueantes = [];
    
    for (var r = 1; r < data.length; r++) {
      var row = data[r];
      var correoSol = (row[colMap["CorreoSolicitante"]] || "").toString().toLowerCase().trim();
      if (correoFiltro && correoSol !== correoFiltro) continue;
      
      var estado = (row[colMap["EstadoSolicitud"]] || "").toString().trim().toUpperCase();
      var clasificacion = (row[colMap["ClasificacionSolicitud"]] || "Anticipo").toString().trim().toUpperCase();
      
      // Aplica a PAGADO (para anticipos) y RECHAZO-CIERRE
      var aplicaEvaluacion = false;
      if (estado === "PAGADO" && clasificacion !== "REINTEGRO") {
        aplicaEvaluacion = true;
      } else if (estado === "RECHAZO-CIERRE") {
        aplicaEvaluacion = true;
      }
      
      if (aplicaEvaluacion) {
        var rawFechaFin = row[colMap["FechaFin"]] || row[colMap["FechaInicio"]] || "";
        var rawFechaMod = row[colMap["FechaModificacion"]] || row[colMap["FechaSolicitud"]] || "";
        
        var fechaFinObj = parsearFechaObj(rawFechaFin);
        fechaFinObj.setHours(0, 0, 0, 0);
        var fechaModObj = parsearFechaObj(rawFechaMod);
        fechaModObj.setHours(0, 0, 0, 0);
        
        // FechaReferencia = max(FechaFin, FechaModificacion)
        var fechaReferencia = fechaFinObj.getTime() > fechaModObj.getTime() ? fechaFinObj : fechaModObj;
        
        var diffMs = hoy.getTime() - fechaReferencia.getTime();
        var diffDias = Math.floor(diffMs / (1000 * 60 * 60 * 24));
        
        if (diffDias >= 15) {
          solicitudesBloqueantes.push({
            idSolicitud: (row[colMap["ID_Solicitud"]] || "").toString().trim(),
            estado: estado,
            clasificacion: clasificacion,
            monto: parseFloat(row[colMap["Monto"]] || 0),
            fechaReferencia: Utilities.formatDate(fechaReferencia, Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy"),
            diasTranscurridos: diffDias
          });
        }
      }
    }
    
    if (solicitudesBloqueantes.length > 0) {
      return {
        status: 'OK',
        bloqueado: true,
        motivo: "No puedes crear nuevas solicitudes de viáticos debido a que tienes solicitudes pendientes de liquidar con más de 15 días de antigüedad. Por favor solventa o liquida tus solicitudes pendientes en 'Cierre de solicitudes' o 'Solicitudes rechazadas' antes de continuar.",
        solicitudesBloqueantes: solicitudesBloqueantes
      };
    }
    
    return { status: 'OK', bloqueado: false, solicitudesBloqueantes: [] };
  } catch (error) {
    Logger.log("Error en verificarBloqueoSolicitudS1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al verificar bloqueo: ' + error.toString() };
  }
}

/**
 * Trigger diario (medianoche): Transiciona solicitudes pre-pago en rechazo con >= 15 días de inactividad
 * a estado CANCELADO, ActorActual = "" y libera el presupuesto en DimConsumo/DimDisponible si correspondía.
 */
function verificarVencimientoSolicitudesRechazadas() {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    if (data.length < 2) return { status: 'OK', canceladas: 0 };
    
    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;
    
    var hoy = new Date();
    hoy.setHours(0, 0, 0, 0);
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    
    var estadosPrePagoRechazo = [
      "RECHAZO-PRESUPUESTO",
      "RECHAZO-PROVISION 1",
      "RECHAZO-PROVISION 2",
      "RECHAZO-PAGO 1"
    ];
    
    var canceladasCount = 0;
    
    for (var r = 1; r < data.length; r++) {
      var row = data[r];
      var estado = (row[colMap["EstadoSolicitud"]] || "").toString().trim().toUpperCase();
      
      var esPrePagoRechazo = false;
      for (var e = 0; e < estadosPrePagoRechazo.length; e++) {
        if (estado === estadosPrePagoRechazo[e]) {
          esPrePagoRechazo = true;
          break;
        }
      }
      
      if (esPrePagoRechazo) {
        var rawFechaMod = row[colMap["FechaModificacion"]] || row[colMap["FechaSolicitud"]] || "";
        var fechaModObj = parsearFechaObj(rawFechaMod);
        fechaModObj.setHours(0, 0, 0, 0);
        
        var diffMs = hoy.getTime() - fechaModObj.getTime();
        var diffDias = Math.floor(diffMs / (1000 * 60 * 60 * 24));
        
        if (diffDias >= 15) {
          var filaIndex = r + 1;
          var montoNum = parseFloat(row[colMap["Monto"]] || 0);
          var centroCostoNum = (row[colMap["CentroCosto"]] || "").toString().split('-')[0].trim() || "1";
          var rubroCodigo = (row[colMap["RubroContable"]] || "8110050400000").toString().trim();
          var fechaSol = (row[colMap["FechaInicio"]] || row[colMap["FechaSolicitud"]] || "").toString();
          var dentroPresupuesto = (row[colMap["DentroPresupuesto"]] || "").toString().trim().toUpperCase();
          
          // Liberación de presupuesto si retenía saldo
          if (estado === "RECHAZO-PROVISION 2" || estado === "RECHAZO-PAGO 1" || dentroPresupuesto === "SI") {
            revertirAfectacionPresupuesto(centroCostoNum, rubroCodigo, fechaSol, montoNum);
          }
          
          // Transicionar a CANCELADO y ActorActual = ""
          sheetFact.getRange(filaIndex, colMap["EstadoSolicitud"] + 1).setValue("CANCELADO");
          sheetFact.getRange(filaIndex, colMap["ActorActual"] + 1).setValue("");
          sheetFact.getRange(filaIndex, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
          
          canceladasCount++;
          Logger.log("Solicitud cancelada por inactividad de 15 días: " + row[colMap["ID_Solicitud"]]);
        }
      }
    }
    
    return { status: 'OK', canceladas: canceladasCount };
  } catch (error) {
    Logger.log("Error en verificarVencimientoSolicitudesRechazadas: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al verificar vencimiento: ' + error.toString() };
  }
}

          destinatario: (obtenerColVal(row, colMap, ["Destinatario", "NombreSolicitante"], "") || "").toString().trim()

  } catch (error) {
    Logger.log("Error en obtenerSolicitudesRechazadas: " + error.toString());

  }

      montoReintegro: obtenerColVal(row, colMap, ["MontoReintegro"], ""),
      fechaReintegro: formatearFechaTexto(obtenerColVal(row, colMap, ["FechaReintegro"], ""), false),
      fechaCierreS: formatearFechaTexto(obtenerColVal(row, colMap, ["FechaCierreS"], ""), true),
      esEditado: obtenerColVal(row, colMap, ["EsEditado"], "No")
    };
    
    detalle.solicitud = detalle;
    detalle.data = detalle;
    return detalle;
  } catch (error) {
    Logger.log("Error en obtenerDetalleSolicitudRechazada: " + error.toString());
    return { status: 'ERROR', success: false, mensaje: 'Error al consultar detalle de solicitud rechazada: ' + error.toString() };
  }
}

/**
 * Handler centralizado para resolver los 5 tipos de rechazo en S2.1:
 * 1. RECHAZO-PRESUPUESTO (Justificación extraordinaria y asignación de ruta 9-16)
 * 2. RECHAZO-PROVISION 1 (Reinicio y revalidación presupuestaria)
 * 3. RECHAZO-PROVISION 2 (Información faltante, directo a Compras)
 * 4. RECHAZO-PAGO 1 (Rechazo de Tesorería, limpia provisión y envía a Compras)
 * 5. RECHAZO-CIERRE (Reenvío de liquidación hacia Compras)
 */
function resolverRechazoS2_1(idSolicitud, tipoRechazo, payload, correoUsuario) {
  try {
    if (!idSolicitud) return { status: 'ERROR', success: false, mensaje: 'ID de solicitud no proporcionado.' };
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES") || obtenerParametro("ID_DB_TRANSACCIONAL");
    if (!idDbTransaccional) return { status: 'ERROR', success: false, mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var idTarget = (idSolicitud || "").toString().trim();
    var filaEncontrada = -1;
    var idxId = obtenerIndiceColumna(colMap, "ID_Solicitud", 0);
    
    for (var r = 1; r < data.length; r++) {
      if ((data[r][idxId] || "").toString().trim() === idTarget) {
        filaEncontrada = r + 1; // 1-indexed para getRange
        break;
      }
    }
    
    // Escaneo lineal de respaldo
    if (filaEncontrada === -1) {
      for (var r2 = 1; r2 < data.length; r2++) {
        for (var c2 = 0; c2 < data[r2].length; c2++) {
          if ((data[r2][c2] || "").toString().trim() === idTarget) {
            filaEncontrada = r2 + 1;
            break;
          }
        }
        if (filaEncontrada !== -1) break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', success: false, mensaje: 'No se encontró la solicitud ' + idSolicitud };
    }
    
    var rowData = data[filaEncontrada - 1];
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var estadoActual = (tipoRechazo || obtenerColVal(rowData, colMap, ["EstadoSolicitud"], "") || "").toString().trim().toUpperCase();
    var nombreSol = obtenerColVal(rowData, colMap, ["NombreSolicitante", "Solicitante"], "Solicitante");
    var correoSol = obtenerColVal(rowData, colMap, ["CorreoSolicitante", "Email"], (correoUsuario || ""));
    var gerencia = obtenerColVal(rowData, colMap, ["Gerencia"], "");
    var cargo = obtenerColVal(rowData, colMap, ["CargoSolicitante", "Cargo"], "");
    var centroCostoStr = obtenerColVal(rowData, colMap, ["CentroCosto"], "");
    var centroCostoNum = (centroCostoStr.split('-')[0] || "").trim() || "1";
    var rubroCodigo = (obtenerColVal(rowData, colMap, ["RubroContable"], "8110050400000") || "8110050400000").toString().trim();
    
    // -------------------------------------------------------------
    // CASO 1: RECHAZO-PRESUPUESTO
    // -------------------------------------------------------------
    if (estadoActual === "RECHAZO-PRESUPUESTO" || estadoActual === "RECHAZADO POR PRESUPUESTO") {
      var justificacionTexto = (payload && payload.justificacion) ? payload.justificacion.toString().trim() : "";
      if (!justificacionTexto) {
        return { status: 'ERROR', success: false, mensaje: 'Debe ingresar el texto de justificación fuera de presupuesto.' };
      }
      
      var montoNum = parseFloat(obtenerColVal(rowData, colMap, ["Monto"], 0)) || 0;
      var ruta = calcularRutaAutorizante({
        Gerencia: gerencia,
        Cargo: cargo,
        Correo: correoSol,
        Region: ""
      }, montoNum, "No");
      
      // Guardar justificación exclusivamente en JustificacionPresupuesto
      establecerColVal(sheetFact, filaEncontrada, colMap, "JustificacionPresupuesto", justificacionTexto, 36);
      establecerColVal(sheetFact, filaEncontrada, colMap, "JustifcacionPresupuesto", justificacionTexto, 56);
      establecerColVal(sheetFact, filaEncontrada, colMap, "FechaModificacion", fechaHoyFormato, 7);
      establecerColVal(sheetFact, filaEncontrada, colMap, "EstadoSolicitud", ruta.estadoInicial || "INICIADO", 3);
      establecerColVal(sheetFact, filaEncontrada, colMap, "DentroPresupuesto", "No", 34);
      establecerColVal(sheetFact, filaEncontrada, colMap, "AutorizacionesPendientes", ruta.autorizacionesPendientes, 4);
      establecerColVal(sheetFact, filaEncontrada, colMap, "ActorActual", ruta.actorActual, 5);
      
      if (ruta.correoJefeRegional) establecerColVal(sheetFact, filaEncontrada, colMap, "CorreoJefeRegional", ruta.correoJefeRegional, 30);
      if (ruta.correoAutorizador1) establecerColVal(sheetFact, filaEncontrada, colMap, "CorreoAutorizador1", ruta.correoAutorizador1, 31);

      SpreadsheetApp.flush();
      
      if (ruta.estadoInicial !== "AUTORIZADO" && ruta.autorizacionesPendientes > 0) {
        var primerCorreo = ruta.actorActual === "Autorizador 0" ? ruta.correoJefeRegional : ruta.correoAutorizador1;
        var primerNivel = ruta.actorActual === "Autorizador 0" ? 0 : 1;
        if (primerCorreo) {
          var datosParaCard = {
            nombre_solicitante: nombreSol,
            correo_solicitante: correoSol,
            cargo_solicitante: cargo,
            gerencia_solicitante: gerencia,
            centro_costo: centroCostoStr,
            agencia: obtenerColVal(rowData, colMap, ["Agencia"], ""),
            duracion_actividad: obtenerColVal(rowData, colMap, ["DuracionActividad"], "Día único"),
            fecha_inicio: obtenerColVal(rowData, colMap, ["FechaInicio"], ""),
            fecha_fin: obtenerColVal(rowData, colMap, ["FechaFin", "FechaInicio"], ""),
            tipo_viatico: obtenerColVal(rowData, colMap, ["TipoViatico"], ""),
            hora_evento: obtenerColVal(rowData, colMap, ["HoraEvento"], ""),
            monto: montoNum,
            monto_total: montoNum,
            motivo_viatico: obtenerColVal(rowData, colMap, ["MotivoViatico"], ""),
            clasificacion_solicitud: obtenerColVal(rowData, colMap, ["ClasificacionSolicitud"], "Anticipo"),
            dentroPresupuesto: "No",
            justificacionPresupuesto: justificacionTexto
          };
          enviarTarjetaEmailAutorizador(primerCorreo, idSolicitud, datosParaCard, primerNivel);
        }
      }
      
      return {
        status: 'OK',
        success: true,
        idSolicitud: idSolicitud,
        estado: ruta.estadoInicial,
        actorActual: ruta.actorActual,
        mensaje: 'Justificación fuera de presupuesto enviada exitosamente.'
      };
    }
    
    // -------------------------------------------------------------
    // CASO 2: RECHAZO-PROVISION 1 (Reinicio de ciclo y revalidación de presupuesto)
    // -------------------------------------------------------------
    else if (estadoActual === "RECHAZO-PROVISION 1") {
      var dViatico = (payload && payload.datosViatico) ? payload.datosViatico : {};
      var montoNuevo = parseFloat(dViatico.monto || dViatico.monto_total || obtenerColVal(rowData, colMap, ["Monto"], 0)) || 0;
      var fechaIniNueva = dViatico.fechaInicio || dViatico.fecha_inicio || obtenerColVal(rowData, colMap, ["FechaInicio"], "") || "";
      var fechaFinNueva = dViatico.fechaFin || dViatico.fecha_fin || fechaIniNueva;
      var esEditadoVal = (dViatico.esEditado === "Si" || dViatico.esEditado === true) ? "Si" : (obtenerColVal(rowData, colMap, ["EsEditado"], "No") || "No");
      
      var archivosActuales = [];
      try {

      } catch (eA) { archivosActuales = []; }
      
      if (payload && payload.listaArchivos && payload.listaArchivos.length > 0) {
        var nuevosGuardados = guardarArchivosEnDrive(idSolicitud, payload.listaArchivos);
        archivosActuales = archivosActuales.concat(nuevosGuardados);
      }
      
      if (dViatico.duracionActividad) establecerColVal(sheetFact, filaEncontrada, colMap, "DuracionActividad", dViatico.duracionActividad, 14);
      if (fechaIniNueva) establecerColVal(sheetFact, filaEncontrada, colMap, "FechaInicio", fechaIniNueva, 15);

      var resPresupuesto = validarYAfectarPresupuesto(centroCostoNum, rubroCodigo, fechaIniNueva, montoNuevo);
      var dentroPresupuesto = (resPresupuesto.dentroPresupuesto || "Si");
      
      if (dentroPresupuesto === "Si") {
        var rutaNueva = calcularRutaAutorizante({
          Gerencia: gerencia,
          Cargo: cargo,
          Correo: correoSol,
          Region: ""
        }, montoNuevo, "Si");
        
        establecerColVal(sheetFact, filaEncontrada, colMap, "EstadoSolicitud", rutaNueva.estadoInicial || "INICIADO", 3);
        establecerColVal(sheetFact, filaEncontrada, colMap, "DentroPresupuesto", "Si", 34);
        establecerColVal(sheetFact, filaEncontrada, colMap, "AutorizacionesPendientes", rutaNueva.autorizacionesPendientes, 4);
        establecerColVal(sheetFact, filaEncontrada, colMap, "ActorActual", rutaNueva.actorActual, 5);
        
        SpreadsheetApp.flush();
        
        if (rutaNueva.estadoInicial !== "AUTORIZADO" && rutaNueva.autorizacionesPendientes > 0) {
          var primerCorreo2 = rutaNueva.actorActual === "Autorizador 0" ? rutaNueva.correoJefeRegional : rutaNueva.correoAutorizador1;
          var primerNivel2 = rutaNueva.actorActual === "Autorizador 0" ? 0 : 1;
          if (primerCorreo2) {
            var datosParaCard2 = {
              nombre_solicitante: nombreSol,
              correo_solicitante: correoSol,
              cargo_solicitante: cargo,
              gerencia_solicitante: gerencia,
              centro_costo: centroCostoStr,
              agencia: obtenerColVal(rowData, colMap, ["Agencia"], ""),
              duracion_actividad: dViatico.duracionActividad || obtenerColVal(rowData, colMap, ["DuracionActividad"], "Día único"),
              fecha_inicio: fechaIniNueva,
              fecha_fin: fechaFinNueva,
              tipo_viatico: dViatico.tipoViatico || obtenerColVal(rowData, colMap, ["TipoViatico"], ""),
              hora_evento: dViatico.horaEvento || obtenerColVal(rowData, colMap, ["HoraEvento"], ""),
              monto: montoNuevo,
              monto_total: montoNuevo,
              motivo_viatico: dViatico.motivoViatico || obtenerColVal(rowData, colMap, ["MotivoViatico"], ""),

              dentroPresupuesto: "Si"
            };
            enviarTarjetaEmailAutorizador(primerCorreo2, idSolicitud, datosParaCard2, primerNivel2);
          }
        }
        
        return {
          status: 'OK',
          success: true,
          idSolicitud: idSolicitud,
          estado: rutaNueva.estadoInicial,
          actorActual: rutaNueva.actorActual,
          mensaje: 'Solicitud reevaluada y enviada a autorización exitosamente.'
        };
      } else {
        // Sin presupuesto
        establecerColVal(sheetFact, filaEncontrada, colMap, "EstadoSolicitud", "RECHAZO-PRESUPUESTO", 3);
        establecerColVal(sheetFact, filaEncontrada, colMap, "DentroPresupuesto", "No", 34);
        establecerColVal(sheetFact, filaEncontrada, colMap, "AutorizacionesPendientes", 0, 4);
        establecerColVal(sheetFact, filaEncontrada, colMap, "ActorActual", "Solicitante", 5);
        
        SpreadsheetApp.flush();
        
        return {
          status: 'OK',
          success: true,
          idSolicitud: idSolicitud,
          estado: 'RECHAZO-PRESUPUESTO',
          actorActual: 'Solicitante',
          mensaje: 'La solicitud actualizada no cuenta con presupuesto suficiente y ha pasado a RECHAZO-PRESUPUESTO.'
        };
      }
    }
    
    // -------------------------------------------------------------
    // CASO 3: RECHAZO-PROVISION 2 (Información faltante, directo a Compras)
    // -------------------------------------------------------------
    else if (estadoActual === "RECHAZO-PROVISION 2") {
      var dViatico = (payload && payload.datosViatico) ? payload.datosViatico : {};
      var esEditadoVal = (dViatico.esEditado === "Si" || dViatico.esEditado === true) ? "Si" : (obtenerColVal(rowData, colMap, ["EsEditado"], "No") || "No");
      
      var archivosActuales = [];
      try {
        var rawA = obtenerColVal(rowData, colMap, ["ArchivosAdjuntos"], "");
        if (rawA) archivosActuales = typeof rawA === 'string' ? JSON.parse(rawA) : rawA;
      } catch (eA) { archivosActuales = []; }
      
      if (payload && payload.listaArchivos && payload.listaArchivos.length > 0) {

      if (dViatico.destinatario) establecerColVal(sheetFact, filaEncontrada, colMap, "Destinatario", dViatico.destinatario, 20);

      for (var c = 0; c < correosCompras.length; c++) {
        try {

            name: "Notificaciones_Viaticos2.0",
            htmlBody: "<p>El solicitante <b>" + nombreSol + "</b> ha completado la información requerida para la solicitud <b>" + idSolicitud + "</b>. La solicitud está lista para provisión en el módulo E1.</p>"
          });
        } catch (eNotif) { Logger.log("Error notificando a compras: " + eNotif.message); }
      }
      
      return {
        status: 'OK',
        success: true,
        idSolicitud: idSolicitud,
        estado: 'AUTORIZADO',
        actorActual: 'Compras',
        mensaje: 'Información solventada exitosamente. La solicitud ha sido devuelta a Compras para provisión.'
      };
    }
    
    // -------------------------------------------------------------
    // CASO 4: RECHAZO-PAGO 1 (Rechazo de Tesorería al Solicitante, resetea provisión y pasa a Compras)
    // -------------------------------------------------------------
    else if (estadoActual === "RECHAZO-PAGO 1") {
      var dViatico = (payload && payload.datosViatico) ? payload.datosViatico : {};

      establecerColVal(sheetFact, filaEncontrada, colMap, "ResolucionProvision", "", 37);

/**
 * Guarda la resolución individual de provisión desde E1.1.
 * Si es Aprobado: guarda ResolucionProvision='Aprobado' y ComentarioProvision (NO altera estado ni actor ni fecha de modificación).
 * Si es Rechazo y requiere reevaluación: EstadoSolicitud='RECHAZO-PROVISION 1', ActorActual='Solicitante', FechaProvision, NombreProvision, libera presupuesto en DimDisponible.
 * Si es Rechazo por información faltante: EstadoSolicitud='RECHAZO-PROVISION 2', ActorActual='Solicitante', FechaProvision, NombreProvision.
 */
function guardarResolucionProvisionE1_1(idSolicitud, resolucion, comentario, nombreUsuario) {
  try {
    if (!resolucion) return { status: 'ERROR', mensaje: 'Debe seleccionar una resolución de provisión.' };
    if (resolucion !== "Aprobado" && (!comentario || comentario.toString().trim() === '')) {
      return { status: 'ERROR', mensaje: 'El comentario es obligatorio para resoluciones de rechazo.' };
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var filaEncontrada = -1;
    var idBuscado = (idSolicitud || "").toString().trim().toUpperCase();
    
    for (var r = 1; r < data.length; r++) {
      var idFila = (data[r][colMap["ID_Solicitud"]] || "").toString().trim().toUpperCase();
      if (idFila === idBuscado) {
        filaEncontrada = r + 1; // 1-indexed para getRange
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la solicitud ' + idSolicitud };
    }
    
    var row = data[filaEncontrada - 1];
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var nombreEditor = nombreUsuario || Session.getActiveUser().getEmail() || "Editor Compras";
    var comentarioTrim = (comentario || "").toString().trim();
    
    // Guardar resolución y comentario en columnas correspondientes
    if (colMap.hasOwnProperty("ResolucionProvision")) {
      sheetFact.getRange(filaEncontrada, colMap["ResolucionProvision"] + 1).setValue(resolucion);
    }
    if (colMap.hasOwnProperty("ComentarioProvision")) {
      sheetFact.getRange(filaEncontrada, colMap["ComentarioProvision"] + 1).setValue(comentarioTrim);
    }
    
    if (resolucion === "Aprobado") {
      // Regla: No modifica FechaModificacion, ActorActual ni EstadoSolicitud
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        mensaje: 'Resolución de provisión registrada como Aprobado. La solicitud ahora está habilitada para ser agrupada en lote contable.'
      };
    } else if (resolucion === "Rechazo y requiere reevaluación") {
      // Liberar presupuesto reservado en DimDisponible / DimConsumo
      var montoNum = parseFloat(row[colMap["Monto"]] || 0);
      var ccTexto = (row[colMap["CentroCosto"]] || "").toString().trim();
      var codigoCC = ccTexto.split('-')[0].trim() || "1";
      var rubroCodigo = (row[colMap["RubroContable"]] || "8110050400000").toString().trim();
      var fechaSol = (row[colMap["FechaSolicitud"]] || fechaHoyFormato).toString();
      
      try {
        revertirAfectacionPresupuesto(codigoCC, rubroCodigo, fechaSol, montoNum);
      } catch (eRev) {
        Logger.log("Aviso al liberar presupuesto en RECHAZO-PROVISION 1: " + eRev.toString());
      }
      
      if (colMap.hasOwnProperty("FechaProvision")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaProvision"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("NombreProvision")) {
        sheetFact.getRange(filaEncontrada, colMap["NombreProvision"] + 1).setValue(nombreEditor);
      }
      if (colMap.hasOwnProperty("FechaModificacion")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("EstadoSolicitud")) {
        sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("RECHAZO-PROVISION 1");
      }
      if (colMap.hasOwnProperty("ActorActual")) {
        sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("Solicitante");
      }
      
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        nuevoEstado: 'RECHAZO-PROVISION 1',
        mensaje: 'Solicitud rechazada y devuelta al Solicitante para reevaluación. El presupuesto reservado ha sido liberado.'
      };
    } else if (resolucion === "Rechazo por información faltante") {
      if (colMap.hasOwnProperty("FechaProvision")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaProvision"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("NombreProvision")) {
        sheetFact.getRange(filaEncontrada, colMap["NombreProvision"] + 1).setValue(nombreEditor);
      }
      if (colMap.hasOwnProperty("FechaModificacion")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("EstadoSolicitud")) {
        sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("RECHAZO-PROVISION 2");
      }
      if (colMap.hasOwnProperty("ActorActual")) {
        sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("Solicitante");
      }
      
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        nuevoEstado: 'RECHAZO-PROVISION 2',
        mensaje: 'Solicitud devuelta al Solicitante para complementar la información faltante.'
      };
    } else {
      return { status: 'ERROR', mensaje: 'Tipo de resolución no reconocida: ' + resolucion };
    }
  } catch (error) {
    Logger.log("Error en guardarResolucionProvisionE1_1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al guardar resolución de provisión: ' + error.toString() };
  }
}

/**
 * Guarda la agrupación en lote de provisión contable para múltiples solicitudes en E1.
 * Guarda AgrupableProvision (JSON), agrega comprobante a ArchivosAdjuntos, FechaProvision, NombreProvision,
 * actualiza FechaModificacion, ActorActual='Tesoreria' y EstadoSolicitud='ENVIADO A PAGO'.
 */
function guardarAgrupacionProvisionE1(idsSolicitudes, datosAsiento, archivoComprobante, nombreUsuario) {
  try {
    if (!idsSolicitudes || !Array.isArray(idsSolicitudes) || idsSolicitudes.length === 0) {
      return { status: 'ERROR', mensaje: 'Debe seleccionar al menos una solicitud para agrupar.' };
    }
    if (!datosAsiento || !datosAsiento.fechaContable || !datosAsiento.fechaValor || !datosAsiento.modulo || !datosAsiento.transaccion || !datosAsiento.codigoRelacion) {
      return { status: 'ERROR', mensaje: 'Complete todos los campos obligatorios del asiento contable.' };
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var nombreEditor = nombreUsuario || Session.getActiveUser().getEmail() || "Editor Compras";
    
    // Subir comprobante a Drive si viene adjunto
    var archivoGuardado = null;
    if (archivoComprobante && (archivoComprobante.ContenidoBase64 || archivoComprobante.DataUrl || archivoComprobante.base64)) {
      var guardados = guardarArchivosEnDrive("PROV_LOTE_" + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "yyyyMMdd_HHmm"), [archivoComprobante]);
      if (guardados && guardados.length > 0) {
        archivoGuardado = guardados[0];
      }
    }
    
    var agrupableJSON = JSON.stringify({
      fechaContable: formatearFechaTexto(datosAsiento.fechaContable, false),
      fechaValor: formatearFechaTexto(datosAsiento.fechaValor, false),
      modulo: datosAsiento.modulo,
      transaccion: datosAsiento.transaccion,
      codigoRelacion: datosAsiento.codigoRelacion,
      crFse: datosAsiento.crFse || ""
    });
    
    var actualizadas = 0;
    
    for (var i = 0; i < idsSolicitudes.length; i++) {
      var idTarget = (idsSolicitudes[i] || "").toString().trim().toUpperCase();
      var filaIndex = -1;
      
      for (var r = 1; r < data.length; r++) {
        var idFila = (data[r][colMap["ID_Solicitud"]] || "").toString().trim().toUpperCase();
        if (idFila === idTarget) {
          filaIndex = r + 1; // 1-indexed para getRange
          break;
        }
      }
      
      if (filaIndex !== -1) {
        // Actualizar AgrupableProvision
        if (colMap.hasOwnProperty("AgrupableProvision")) {
          sheetFact.getRange(filaIndex, colMap["AgrupableProvision"] + 1).setValue(agrupableJSON);
        }
        
        // Agregar archivo al listado de ArchivosAdjuntos
        if (archivoGuardado && colMap.hasOwnProperty("ArchivosAdjuntos")) {
          var archivosPrevios = [];
          try {
            var rawA = data[filaIndex - 1][colMap["ArchivosAdjuntos"]];
            if (rawA) archivosPrevios = typeof rawA === "string" ? JSON.parse(rawA) : rawA;
          } catch (eA) { archivosPrevios = []; }
          archivosPrevios.push(archivoGuardado);
          sheetFact.getRange(filaIndex, colMap["ArchivosAdjuntos"] + 1).setValue(JSON.stringify(archivosPrevios));
        }
        
        if (colMap.hasOwnProperty("FechaProvision")) {
          sheetFact.getRange(filaIndex, colMap["FechaProvision"] + 1).setValue(fechaHoyFormato);
        }
        if (colMap.hasOwnProperty("NombreProvision")) {
          sheetFact.getRange(filaIndex, colMap["NombreProvision"] + 1).setValue(nombreEditor);
        }
        if (colMap.hasOwnProperty("FechaModificacion")) {
          sheetFact.getRange(filaIndex, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
        }
        if (colMap.hasOwnProperty("ActorActual")) {
          sheetFact.getRange(filaIndex, colMap["ActorActual"] + 1).setValue("Tesoreria");
        }
        if (colMap.hasOwnProperty("EstadoSolicitud")) {
          sheetFact.getRange(filaIndex, colMap["EstadoSolicitud"] + 1).setValue("ENVIADO A PAGO");
        }
        
        actualizadas++;
      }
    }
    
    return {
      status: 'OK',
      totalAgrupadas: actualizadas,
      mensaje: 'Se han agrupado y enviado a Tesorería ' + actualizadas + ' solicitudes con estado ENVIADO A PAGO.'
    };
  } catch (error) {
    Logger.log("Error en guardarAgrupacionProvisionE1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al agrupar solicitudes de provisión: ' + error.toString() };
  }
}

/**
 * Guarda la resolución individual de procesamiento de pago desde E2.1.
 * Si es Aprobado: guarda ResolucionProcesamiento='Aprobado' y ComentarioProcesamiento (NO altera estado ni actor ni fecha de modificación).
 * Si es Rechazo hacia solicitante: EstadoSolicitud='RECHAZO-PAGO 1', ActorActual='Solicitante', FechaProcesamiento, NombreProcesamiento, FechaModificacion.
 * Si es Rechazo hacia compras: EstadoSolicitud='RECHAZO-PAGO 2', ActorActual='Compras', FechaProcesamiento, NombreProcesamiento, FechaModificacion.
 */
function guardarResolucionProcesamientoE2_1(idSolicitud, resolucion, comentario, nombreUsuario) {
  try {
    if (!resolucion) return { status: 'ERROR', mensaje: 'Debe seleccionar una resolución de procesamiento.' };
    if (resolucion !== "Aprobado" && (!comentario || comentario.toString().trim() === '')) {
      return { status: 'ERROR', mensaje: 'El comentario es obligatorio para resoluciones de rechazo.' };
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var filaEncontrada = -1;
    var idBuscado = (idSolicitud || "").toString().trim().toUpperCase();
    
    for (var r = 1; r < data.length; r++) {
      var idFila = (data[r][colMap["ID_Solicitud"]] || "").toString().trim().toUpperCase();
      if (idFila === idBuscado) {
        filaEncontrada = r + 1; // 1-indexed para getRange
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la solicitud ' + idSolicitud };
    }
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var nombreEditor = nombreUsuario || Session.getActiveUser().getEmail() || "Editor Tesorería";
    var comentarioTrim = (comentario || "").toString().trim();
    
    // Guardar resolución y comentario
    if (colMap.hasOwnProperty("ResolucionProcesamiento")) {
      sheetFact.getRange(filaEncontrada, colMap["ResolucionProcesamiento"] + 1).setValue(resolucion);
    }
    if (colMap.hasOwnProperty("ComentarioProcesamiento")) {
      sheetFact.getRange(filaEncontrada, colMap["ComentarioProcesamiento"] + 1).setValue(comentarioTrim);
    }
    
    if (resolucion === "Aprobado") {
      // Regla: No modifica FechaModificacion, ActorActual ni EstadoSolicitud hasta agrupar
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        mensaje: 'Resolución de procesamiento registrada como Aprobado. La solicitud ahora está habilitada para ser agrupada en lote de pago bancario.'
      };
    } else if (resolucion === "Rechazo hacia solicitante") {
      if (colMap.hasOwnProperty("FechaProcesamiento")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaProcesamiento"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("NombreProcesamiento")) {
        sheetFact.getRange(filaEncontrada, colMap["NombreProcesamiento"] + 1).setValue(nombreEditor);
      }
      if (colMap.hasOwnProperty("FechaModificacion")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("EstadoSolicitud")) {
        sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("RECHAZO-PAGO 1");
      }
      if (colMap.hasOwnProperty("ActorActual")) {
        sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("Solicitante");
      }
      
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        nuevoEstado: 'RECHAZO-PAGO 1',
        mensaje: 'Solicitud devuelta al Solicitante para aclaración o corrección bancaria.'
      };
    } else if (resolucion === "Rechazo hacia compras") {
      if (colMap.hasOwnProperty("FechaProcesamiento")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaProcesamiento"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("NombreProcesamiento")) {
        sheetFact.getRange(filaEncontrada, colMap["NombreProcesamiento"] + 1).setValue(nombreEditor);
      }
      if (colMap.hasOwnProperty("FechaModificacion")) {
        sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
      }
      if (colMap.hasOwnProperty("EstadoSolicitud")) {
        sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("RECHAZO-PAGO 2");
      }
      if (colMap.hasOwnProperty("ActorActual")) {
        sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("Compras");
      }
      
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        nuevoEstado: 'RECHAZO-PAGO 2',
        mensaje: 'Solicitud devuelta a la bandeja de Compras para ajuste de provisión contable.'
      };
    } else {
      return { status: 'ERROR', mensaje: 'Tipo de resolución no reconocida: ' + resolucion };
    }
  } catch (error) {
    Logger.log("Error en guardarResolucionProcesamientoE2_1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al guardar resolución de procesamiento: ' + error.toString() };
  }
}

/**
 * Guarda la agrupación en lote de pago bancario para múltiples solicitudes en E2.
 * Guarda AgrupableProcesamiento (JSON), anexa comprobante a ArchivosAdjuntos, FechaProcesamiento, NombreProcesamiento,
 * actualiza FechaModificacion y realiza la transición de estado:
 * - Anticipo => EstadoSolicitud='PAGADO', ActorActual='Solicitante'
 * - Reintegro => EstadoSolicitud='FINALIZADO', ActorActual=''
 */
function guardarAgrupacionProcesamientoE2(idsSolicitudes, datosAsiento, archivoComprobante, nombreUsuario) {
  try {
    if (!idsSolicitudes || !Array.isArray(idsSolicitudes) || idsSolicitudes.length === 0) {
      return { status: 'ERROR', mensaje: 'Debe seleccionar al menos una solicitud para agrupar.' };
    }
    if (!datosAsiento || !datosAsiento.fechaContable || !datosAsiento.fechaValor || !datosAsiento.modulo || !datosAsiento.transaccion || !datosAsiento.codigoRelacion) {
      return { status: 'ERROR', mensaje: 'Complete todos los campos obligatorios del asiento contable.' };
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var nombreEditor = nombreUsuario || Session.getActiveUser().getEmail() || "Editor Tesorería";
    
    // Subir comprobante bancario a Drive
    var archivoGuardado = null;
    if (archivoComprobante && (archivoComprobante.ContenidoBase64 || archivoComprobante.DataUrl || archivoComprobante.base64)) {
      var guardados = guardarArchivosEnDrive("PAGO_LOTE_" + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "yyyyMMdd_HHmm"), [archivoComprobante]);
      if (guardados && guardados.length > 0) {
        archivoGuardado = guardados[0];
      }
    }
    
    var agrupableJSON = JSON.stringify({
      fechaContable: formatearFechaTexto(datosAsiento.fechaContable, false),
      fechaValor: formatearFechaTexto(datosAsiento.fechaValor, false),
      modulo: datosAsiento.modulo,
      transaccion: datosAsiento.transaccion,
      codigoRelacion: datosAsiento.codigoRelacion,
      crFse: datosAsiento.crFse || "",
      fechaPago: formatearFechaTexto(datosAsiento.fechaPago || datosAsiento.fechaValor, false)
    });
    
    var actualizadas = 0;
    
    for (var i = 0; i < idsSolicitudes.length; i++) {
      var idTarget = (idsSolicitudes[i] || "").toString().trim().toUpperCase();
      var filaIndex = -1;
      
      for (var r = 1; r < data.length; r++) {
        var idFila = (data[r][colMap["ID_Solicitud"]] || "").toString().trim().toUpperCase();
        if (idFila === idTarget) {
          filaIndex = r + 1; // 1-indexed para getRange
          break;
        }
      }
      
      if (filaIndex !== -1) {
        var rowVal = data[filaIndex - 1];
        var clasificacion = (rowVal[colMap["ClasificacionSolicitud"]] || "Anticipo").toString().trim();
        
        // Escribir AgrupableProcesamiento
        if (colMap.hasOwnProperty("AgrupableProcesamiento")) {
          sheetFact.getRange(filaIndex, colMap["AgrupableProcesamiento"] + 1).setValue(agrupableJSON);
        }
        
        // Anexar archivo
        if (archivoGuardado && colMap.hasOwnProperty("ArchivosAdjuntos")) {
          var archivosPrevios = [];
          try {
            var rawA = rowVal[colMap["ArchivosAdjuntos"]];
            if (rawA) archivosPrevios = typeof rawA === "string" ? JSON.parse(rawA) : rawA;
          } catch (eA) { archivosPrevios = []; }
          archivosPrevios.push(archivoGuardado);
          sheetFact.getRange(filaIndex, colMap["ArchivosAdjuntos"] + 1).setValue(JSON.stringify(archivosPrevios));
        }
        
        if (colMap.hasOwnProperty("FechaProcesamiento")) {
          sheetFact.getRange(filaIndex, colMap["FechaProcesamiento"] + 1).setValue(fechaHoyFormato);
        }
        if (colMap.hasOwnProperty("NombreProcesamiento")) {
          sheetFact.getRange(filaIndex, colMap["NombreProcesamiento"] + 1).setValue(nombreEditor);
        }
        if (colMap.hasOwnProperty("FechaModificacion")) {
          sheetFact.getRange(filaIndex, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
        }
        
        if (clasificacion.toLowerCase() === "reintegro") {
          if (colMap.hasOwnProperty("EstadoSolicitud")) {
            sheetFact.getRange(filaIndex, colMap["EstadoSolicitud"] + 1).setValue("FINALIZADO");
          }
          if (colMap.hasOwnProperty("ActorActual")) {
            sheetFact.getRange(filaIndex, colMap["ActorActual"] + 1).setValue("");
          }
        } else {
          if (colMap.hasOwnProperty("EstadoSolicitud")) {
            sheetFact.getRange(filaIndex, colMap["EstadoSolicitud"] + 1).setValue("PAGADO");
          }
          if (colMap.hasOwnProperty("ActorActual")) {

/**
 * Guarda la liquidación / cierre enviada por el solicitante en S3.1.
 * Actualiza:
 * - ArchivosAdjuntos: agrega nuevos comprobantes/facturas subidos a Google Drive.
 * - TipoCierre: "Solo cierre" o "Reintegro y cierre".
 * - MontoReintegro y FechaReintegro (si aplica).
 * - FechaCierreS = fecha actual.
 * - FechaModificacion = fecha actual.
 * - ActorActual = "Compras".
 * - EstadoSolicitud = "LIQUIDADO".
 */
function guardarCierreSolicitudS3_1(idSolicitud, datosCierre, listaArchivos, nombreUsuario) {
  try {
    if (!idSolicitud) return { status: 'ERROR', mensaje: 'ID de solicitud no especificado.' };
    if (!datosCierre || !datosCierre.tipoCierre) {
      return { status: 'ERROR', mensaje: 'Debe seleccionar el Tipo de Cierre.' };
    }
    
    var tipoCierre = datosCierre.tipoCierre.toString().trim();
    if (tipoCierre === "Reintegro y cierre") {
      if (!datosCierre.montoReintegro || isNaN(parseFloat(datosCierre.montoReintegro)) || parseFloat(datosCierre.montoReintegro) <= 0) {
        return { status: 'ERROR', mensaje: 'Debe ingresar un monto válido a reintegrar.' };
      }
      if (!datosCierre.fechaReintegro || datosCierre.fechaReintegro.toString().trim() === '') {
        return { status: 'ERROR', mensaje: 'Debe ingresar la fecha del reintegro.' };
      }
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var filaEncontrada = -1;
    for (var r = 1; r < data.length; r++) {
      if ((data[r][colMap["ID_Solicitud"]] || "").toString().trim().toUpperCase() === idSolicitud.toString().trim().toUpperCase()) {
        filaEncontrada = r + 1; // 1-indexed
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la solicitud ' + idSolicitud };
    }
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    
    // Subir comprobantes/facturas a Drive
    if (listaArchivos && Array.isArray(listaArchivos) && listaArchivos.length > 0) {
      var guardados = guardarArchivosEnDrive(idSolicitud + "_LIQUIDACION", listaArchivos);
      if (guardados && guardados.length > 0) {
        var archivosPrevios = [];
        try {
          var rawA = data[filaEncontrada - 1][colMap["ArchivosAdjuntos"]];
          if (rawA) archivosPrevios = typeof rawA === "string" ? JSON.parse(rawA) : rawA;
        } catch (eA) { archivosPrevios = []; }
        
        for (var a = 0; a < guardados.length; a++) {
          archivosPrevios.push(guardados[a]);
        }
        if (colMap.hasOwnProperty("ArchivosAdjuntos")) {
          sheetFact.getRange(filaEncontrada, colMap["ArchivosAdjuntos"] + 1).setValue(JSON.stringify(archivosPrevios));
        }
      }
    }
    
    // Guardar datos de cierre
    if (colMap.hasOwnProperty("TipoCierre")) {
      sheetFact.getRange(filaEncontrada, colMap["TipoCierre"] + 1).setValue(tipoCierre);
    }
    if (colMap.hasOwnProperty("MontoReintegro")) {
      var mReint = (tipoCierre === "Reintegro y cierre") ? parseFloat(datosCierre.montoReintegro) : "";
      sheetFact.getRange(filaEncontrada, colMap["MontoReintegro"] + 1).setValue(mReint);
    }
    if (colMap.hasOwnProperty("FechaReintegro")) {
      var fReint = (tipoCierre === "Reintegro y cierre") ? formatearFechaTexto(datosCierre.fechaReintegro, false) : "";
      sheetFact.getRange(filaEncontrada, colMap["FechaReintegro"] + 1).setValue(fReint);
    }
    if (colMap.hasOwnProperty("FechaCierreS")) {
      sheetFact.getRange(filaEncontrada, colMap["FechaCierreS"] + 1).setValue(fechaHoyFormato);
    }
    
    // Actualizar estado y actor
    if (colMap.hasOwnProperty("FechaModificacion")) {
      sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
    }
    if (colMap.hasOwnProperty("ActorActual")) {
      sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("Compras");
    }
    if (colMap.hasOwnProperty("EstadoSolicitud")) {
      sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("LIQUIDADO");
    }
    
    return {
      status: 'OK',
      idSolicitud: idSolicitud,
      nuevoEstado: 'LIQUIDADO',
      mensaje: 'Liquidación de viático registrada exitosamente. La solicitud ha pasado a revisión por el área de Compras (Estado: LIQUIDADO).'
    };
  } catch (error) {
    Logger.log("Error en guardarCierreSolicitudS3_1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al procesar cierre de solicitud: ' + error.toString() };
  }
}

      comentarioProvision: colMap.hasOwnProperty("ComentarioProvision") ? (row[colMap["ComentarioProvision"]] || "") : "",
      // Auditoría de Procesamiento (Tesorería)
      nombreProcesamiento: colMap.hasOwnProperty("NombreProcesamiento") ? (row[colMap["NombreProcesamiento"]] || "") : "",
      fechaProcesamiento: formatearFechaTexto(row[colMap["FechaProcesamiento"]], true),
      comentarioProcesamiento: colMap.hasOwnProperty("ComentarioProcesamiento") ? (row[colMap["ComentarioProcesamiento"]] || "") : "",
      // Auditoría de Cierre en Compras previa
      resolucionCierreE: colMap.hasOwnProperty("ResolucionCierreE") ? (row[colMap["ResolucionCierreE"]] || "") : "",
      comentarioCierreE: colMap.hasOwnProperty("ComentarioCierreE") ? (row[colMap["ComentarioCierreE"]] || "") : "",
      fechaCierreE: formatearFechaTexto(row[colMap["FechaCierreE"]], true),
      nombreCierreE: colMap.hasOwnProperty("NombreCierreE") ? (row[colMap["NombreCierreE"]] || "") : ""
    };
  } catch (error) {
    Logger.log("Error en obtenerDetalleCierreE3_1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al consultar detalle de cierre: ' + error.toString() };
  }
}

/**
 * Guarda la resolución de revisión de cierre emitida por Compras en E3.1.
 * - Si es Aprobado y TipoCierre="Solo cierre": pasa a FINALIZADO, ActorActual="", FechaCierreE, NombreCierreE, FechaModificacion.
 * - Si es Aprobado y TipoCierre="Reintegro y cierre": guarda ResolucionCierreE="Aprobado", ComentarioCierreE (queda habilitada para agrupar en E3).
 * - Si es Rechazado: pasa a RECHAZO-CIERRE, ActorActual="Solicitante", FechaCierreE, NombreCierreE, FechaModificacion.
 */
function guardarResolucionCierreE3_1(idSolicitud, resolucionODatos, comentarioONombre, optNombre) {
  try {
    var resolucion = '';
    var comentario = '';
    var nombreUsuario = '';
    
    if (typeof resolucionODatos === 'object' && resolucionODatos !== null) {
      resolucion = (resolucionODatos.resolucion || resolucionODatos.resolucionCierreE || '').toString().trim();
      comentario = (resolucionODatos.comentario || resolucionODatos.comentarioCierreE || '').toString().trim();
      nombreUsuario = comentarioONombre || (resolucionODatos.nombreUsuario || '');
    } else {
      resolucion = (resolucionODatos || '').toString().trim();
      comentario = (comentarioONombre || '').toString().trim();
      nombreUsuario = optNombre || '';
    }
    
    if (!resolucion) return { status: 'ERROR', mensaje: 'Debe seleccionar una resolución de cierre.' };
    if (resolucion === "Rechazado" && (!comentario || comentario.toString().trim() === '')) {
      return { status: 'ERROR', mensaje: 'El comentario es obligatorio para resoluciones de rechazo.' };
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var filaEncontrada = -1;
    for (var r = 1; r < data.length; r++) {
      if ((data[r][colMap["ID_Solicitud"]] || "").toString().trim() === idSolicitud.toString().trim()) {
        filaEncontrada = r + 1; // 1-indexed
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { status: 'ERROR', mensaje: 'No se encontró la solicitud ' + idSolicitud };
    }
    
    var rowVal = data[filaEncontrada - 1];
    var tipoCierre = (rowVal[colMap["TipoCierre"]] || "").toString().trim();
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var nombreEditor = nombreUsuario || Session.getActiveUser().getEmail() || "Editor Compras";
    var comentarioTrim = (comentario || "").toString().trim();
    
    // Guardar resolución y comentario
    if (colMap.hasOwnProperty("ResolucionCierreE")) {
      sheetFact.getRange(filaEncontrada, colMap["ResolucionCierreE"] + 1).setValue(resolucion);
    }
    if (colMap.hasOwnProperty("ComentarioCierreE")) {
      sheetFact.getRange(filaEncontrada, colMap["ComentarioCierreE"] + 1).setValue(comentarioTrim);
    }
    
    if (resolucion === "Aprobado") {
      if (tipoCierre.toLowerCase() === "solo cierre") {
        // Cierre directo sin asiento de reintegro
        sheetFact.getRange(filaEncontrada, colMap["FechaCierreE"] + 1).setValue(fechaHoyFormato);
        sheetFact.getRange(filaEncontrada, colMap["NombreCierreE"] + 1).setValue(nombreEditor);
        sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
        sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("FINALIZADO");
        sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("");
        
        return {
          status: 'OK',
          idSolicitud: idSolicitud,
          resolucion: resolucion,
          nuevoEstado: 'FINALIZADO',
          mensaje: 'Liquidación aprobada con éxito. El expediente de viático ha sido completado y finalizado (Estado: FINALIZADO).'
        };
      } else {
        // Reintegro y cierre: Queda aprobada pero pendiente de asiento contable en E3
        sheetFact.getRange(filaEncontrada, colMap["FechaCierreE"] + 1).setValue(fechaHoyFormato);
        sheetFact.getRange(filaEncontrada, colMap["NombreCierreE"] + 1).setValue(nombreEditor);
        
        return {
          status: 'OK',
          idSolicitud: idSolicitud,
          resolucion: resolucion,
          nuevoEstado: 'LIQUIDADO',
          mensaje: 'Resolución de cierre registrada como Aprobado. La solicitud ha quedado habilitada para asentar el reintegro contable en lote en la bandeja E3.'
        };
      }
    } else if (resolucion === "Rechazado") {
      sheetFact.getRange(filaEncontrada, colMap["FechaCierreE"] + 1).setValue(fechaHoyFormato);
      sheetFact.getRange(filaEncontrada, colMap["NombreCierreE"] + 1).setValue(nombreEditor);
      sheetFact.getRange(filaEncontrada, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
      sheetFact.getRange(filaEncontrada, colMap["EstadoSolicitud"] + 1).setValue("RECHAZO-CIERRE");
      sheetFact.getRange(filaEncontrada, colMap["ActorActual"] + 1).setValue("Solicitante");
      
      return {
        status: 'OK',
        idSolicitud: idSolicitud,
        resolucion: resolucion,
        nuevoEstado: 'RECHAZO-CIERRE',
        mensaje: 'Liquidación rechazada y devuelta al Solicitante para corrección o aclaración de comprobantes.'
      };
    } else {
      return { status: 'ERROR', mensaje: 'Tipo de resolución no reconocida: ' + resolucion };
    }
  } catch (error) {
    Logger.log("Error en guardarResolucionCierreE3_1: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al guardar resolución de cierre: ' + error.toString() };
  }
}

/**
 * Guarda la agrupación en lote de reintegros de liquidación en E3.
 * Guarda AgrupableCierreE (JSON), anexa comprobante a ArchivosAdjuntos, FechaCierreE, NombreCierreE,
 * actualiza FechaModificacion, EstadoSolicitud='FINALIZADO' y ActorActual=''.
 */
function guardarAgrupacionCierreE3(idsSolicitudes, datosAsiento, archivoComprobante, nombreUsuario) {
  try {
    if (!idsSolicitudes || !Array.isArray(idsSolicitudes) || idsSolicitudes.length === 0) {
      return { status: 'ERROR', mensaje: 'Debe seleccionar al menos una solicitud para agrupar.' };
    }
    if (!datosAsiento || !datosAsiento.fechaContable || !datosAsiento.fechaValor || !datosAsiento.modulo || !datosAsiento.transaccion || !datosAsiento.codigoRelacion) {
      return { status: 'ERROR', mensaje: 'Complete todos los campos obligatorios del asiento contable.' };
    }
    
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };
    
    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();
    
    var headers = data[0];
    var colMap = obtenerColMapTransaccional(headers);
    
    var fechaHoyFormato = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "dd/MM/yyyy HH:mm");
    var nombreEditor = nombreUsuario || Session.getActiveUser().getEmail() || "Editor Compras";
    
    // Subir comprobante contable de cierre a Drive
    var archivoGuardado = null;
    if (archivoComprobante && (archivoComprobante.ContenidoBase64 || archivoComprobante.DataUrl || archivoComprobante.base64)) {
      var guardados = guardarArchivosEnDrive("CIERRE_LOTE_" + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "GMT-6", "yyyyMMdd_HHmm"), [archivoComprobante]);
      if (guardados && guardados.length > 0) {
        archivoGuardado = guardados[0];
      }
    }
    
    var agrupableJSON = JSON.stringify({
      fechaContable: datosAsiento.fechaContable,
      fechaValor: datosAsiento.fechaValor,
      modulo: datosAsiento.modulo,
      transaccion: datosAsiento.transaccion,
      codigoRelacion: datosAsiento.codigoRelacion,
      crFse: datosAsiento.crFse || ""
    });
    
    var actualizadas = 0;
    
    for (var i = 0; i < idsSolicitudes.length; i++) {
      var idTarget = (idsSolicitudes[i] || "").toString().trim();
      var filaIndex = -1;
      
      for (var r = 1; r < data.length; r++) {
        if ((data[r][colMap["ID_Solicitud"]] || "").toString().trim() === idTarget) {
          filaIndex = r + 1; // 1-indexed
          break;
        }
      }
      
      if (filaIndex !== -1) {
        // Escribir AgrupableCierreE
        if (colMap.hasOwnProperty("AgrupableCierreE")) {
          sheetFact.getRange(filaIndex, colMap["AgrupableCierreE"] + 1).setValue(agrupableJSON);
        }
        
        // Anexar archivo a ArchivosAdjuntos sin sobreescribir los previos
        if (archivoGuardado) {
          var archivosPrevios = [];
          try {
            var rawA = data[filaIndex - 1][colMap["ArchivosAdjuntos"]];
            if (rawA) archivosPrevios = JSON.parse(rawA);
          } catch (eA) { archivosPrevios = []; }
          archivosPrevios.push(archivoGuardado);
          sheetFact.getRange(filaIndex, colMap["ArchivosAdjuntos"] + 1).setValue(JSON.stringify(archivosPrevios));
        }
        
        sheetFact.getRange(filaIndex, colMap["FechaCierreE"] + 1).setValue(fechaHoyFormato);
        sheetFact.getRange(filaIndex, colMap["NombreCierreE"] + 1).setValue(nombreEditor);
        sheetFact.getRange(filaIndex, colMap["FechaModificacion"] + 1).setValue(fechaHoyFormato);
        sheetFact.getRange(filaIndex, colMap["EstadoSolicitud"] + 1).setValue("FINALIZADO");
        sheetFact.getRange(filaIndex, colMap["ActorActual"] + 1).setValue("");
        
        actualizadas++;
      }
    }
    
    return {
      status: 'OK',
      solicitudesActualizadas: actualizadas,
      mensaje: 'Se completó la agrupación contable de cierre para ' + actualizadas + ' solicitud(es). Estado actualizado a FINALIZADO.'
    };
  } catch (error) {
    Logger.log("Error en guardarAgrupacionCierreE3: " + error.toString());
    return { status: 'ERROR', mensaje: 'Error al procesar la agrupación contable de cierre: ' + error.toString() };
  }
}

/**
 * Obtiene todas las solicitudes en estados activos / en curso (no terminales).
 * Filtra por correo para rol SOLICITANTE, o devuelve todas para roles operativos / autorizadores / admin.
 * Estados excluidos: FINALIZADO, CANCELADO, RECHAZO-AUTORIZACION.
 */
function obtenerSolicitudesEnCurso(correoUsuario, rolUsuario) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };

    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();

    if (data.length <= 1) return { status: 'OK', solicitudes: [] };

    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;

    var emailActivo = (correoUsuario || Session.getActiveUser().getEmail() || "").toLowerCase().trim();
    var rolActivo = (rolUsuario || "SOLICITANTE").toString().toUpperCase().trim();
    var esSolicitantePuro = (rolActivo === "SOLICITANTE");

    var estadosTerminales = ["FINALIZADO", "CANCELADO", "RECHAZO-AUTORIZACION"];
    var resultado = [];

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var estado = (row[colMap["EstadoSolicitud"]] || "").toString().trim().toUpperCase();
      if (!estado || estadosTerminales.indexOf(estado) !== -1) {
        continue;
      }

      var correoSolFila = (row[colMap["CorreoSolicitante"]] || "").toString().toLowerCase().trim();
      if (esSolicitantePuro && correoSolFila !== emailActivo) {
        continue;
      }

      var idSol = (row[colMap["ID_Solicitud"]] || "").toString().trim();
      var solicitante = (row[colMap["NombreSolicitante"]] || "").toString().trim();
      var monto = parseFloat(row[colMap["MontoTotal"]] || 0);
      var codigoCC = (row[colMap["CentroCosto"]] || "").toString().trim();
      var fechaSol = (row[colMap["FechaSolicitud"]] || "").toString().trim();
      var fechaMod = (row[colMap["FechaModificacion"]] || "").toString().trim();
      var actorActual = (row[colMap["ActorActual"]] || "").toString().trim();
      var tipoViatico = (row[colMap["TipoViatico"]] || "").toString().trim();
      var clasificacion = (row[colMap["ClasificacionSolicitud"]] || "Anticipo").toString().trim();
      var esEditado = (row[colMap["EsEditado"]] || "No").toString().trim();

      resultado.push({
        idSolicitud: idSol,
        solicitante: solicitante,
        correoSolicitante: correoSolFila,
        monto: monto,
        montoFormatted: "$ " + monto.toFixed(2),
        codigoCC: codigoCC,
        fechaSolicitud: fechaSol,
        fechaModificacion: fechaMod,
        estadoSolicitud: estado,
        actorActual: actorActual,
        tipoViatico: tipoViatico,
        clasificacionSolicitud: clasificacion,
        esEditado: esEditado
      });
    }

    return {
      status: 'OK',
      solicitudes: resultado
    };
  } catch (e) {
    Logger.log("Error en obtenerSolicitudesEnCurso: " + e.toString());
    return { status: 'ERROR', mensaje: e.toString() };
  }
}

  }
}

/**
 * Obtiene todas las solicitudes en estados terminales (concluidas definitivamente).
 * Estados admitidos: FINALIZADO, CANCELADO, RECHAZO-AUTORIZACION.
 * Filtra por correo para rol SOLICITANTE, o devuelve todas para roles operativos / autorizadores / admin.
 */
function obtenerSolicitudesHistorico(correoUsuario, rolUsuario) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES en Parametros_Viaticos2.0.' };

    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();

    if (data.length <= 1) return { status: 'OK', solicitudes: [] };

    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;

    var emailActivo = (correoUsuario || Session.getActiveUser().getEmail() || "").toLowerCase().trim();
    var rolActivo = (rolUsuario || "SOLICITANTE").toString().toUpperCase().trim();
    var esSolicitantePuro = (rolActivo === "SOLICITANTE");

    var estadosTerminales = ["FINALIZADO", "CANCELADO", "RECHAZO-AUTORIZACION"];
    var resultado = [];

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var estado = (row[colMap["EstadoSolicitud"]] || "").toString().trim().toUpperCase();
      if (!estado || estadosTerminales.indexOf(estado) === -1) {
        continue;
      }

      var correoSolFila = (row[colMap["CorreoSolicitante"]] || "").toString().toLowerCase().trim();
      if (esSolicitantePuro && correoSolFila !== emailActivo) {
        continue;
      }

      var idSol = (row[colMap["ID_Solicitud"]] || "").toString().trim();
      var solicitante = (row[colMap["NombreSolicitante"]] || "").toString().trim();
      var monto = parseFloat(row[colMap["MontoTotal"]] || 0);
      var codigoCC = (row[colMap["CentroCosto"]] || "").toString().trim();
      var fechaSol = (row[colMap["FechaSolicitud"]] || "").toString().trim();
      var fechaMod = (row[colMap["FechaModificacion"]] || "").toString().trim();
      var tipoViatico = (row[colMap["TipoViatico"]] || "").toString().trim();
      var clasificacion = (row[colMap["ClasificacionSolicitud"]] || "Anticipo").toString().trim();
      var esEditado = (row[colMap["EsEditado"]] || "No").toString().trim();

      resultado.push({
        idSolicitud: idSol,
        solicitante: solicitante,
        correoSolicitante: correoSolFila,
        monto: monto,
        montoFormatted: "$ " + monto.toFixed(2),
        codigoCC: codigoCC,
        fechaSolicitud: fechaSol,
        fechaModificacion: fechaMod,
        estadoSolicitud: estado,
        tipoViatico: tipoViatico,
        clasificacionSolicitud: clasificacion,
        esEditado: esEditado
      });
    }

    return {
      status: 'OK',
      solicitudes: resultado
    };
  } catch (e) {
    Logger.log("Error en obtenerSolicitudesHistorico: " + e.toString());
    return { status: 'ERROR', mensaje: e.toString() };
  }
}

/**
 * Obtiene el expediente inmutable completo de una solicitud en el histórico.
 */
function obtenerExpedienteHistorico(idSolicitud) {
  try {
    var idDbTransaccional = obtenerParametro("ID_DB_TRANSACCIONES");
    if (!idDbTransaccional) return { status: 'ERROR', mensaje: 'No se encontró ID_DB_TRANSACCIONES.' };

    var ssViaticos = SpreadsheetApp.openById(idDbTransaccional);
    var sheetFact = ssViaticos.getSheetByName("DimTransaccional") || ssViaticos.getSheets()[0];
    var data = sheetFact.getDataRange().getValues();

    var headers = data[0];
    var colMap = {};
    for (var h = 0; h < headers.length; h++) colMap[headers[h].toString().trim()] = h;

    var row = null;
    var target = (idSolicitud || "").toString().trim();
    for (var i = 1; i < data.length; i++) {
      if ((data[i][colMap["ID_Solicitud"]] || "").toString().trim() === target) {
        row = data[i];
        break;
      }
    }

    if (!row) return { status: 'ERROR', mensaje: 'Solicitud histórica no encontrada.' };

    var archivos = [];
    try {
      var rawArch = row[colMap["ArchivosAdjuntos"]];
      if (rawArch) archivos = JSON.parse(rawArch);
    } catch(eA) { archivos = []; }

    var autorizaciones = [];
    try {
      var rawAut = row[colMap["DatosAutorizacion"]];
      if (rawAut) autorizaciones = JSON.parse(rawAut);
    } catch(eAu) { autorizaciones = []; }

    var agrProv = null;
    try {
      var rawP = row[colMap["AgrupableProvision"]];
      if (rawP) agrProv = JSON.parse(rawP);
    } catch(eP) { agrProv = null; }

    var agrProc = null;
    try {
      var rawPr = row[colMap["AgrupableProcesamiento"]];
      if (rawPr) agrProc = JSON.parse(rawPr);
    } catch(ePr) { agrProc = null; }

    var agrCierre = null;
    try {
      var rawC = row[colMap["AgrupableCierreE"]];
      if (rawC) agrCierre = JSON.parse(rawC);
    } catch(eC) { agrCierre = null; }

    var montoNum = parseFloat(row[colMap["MontoTotal"]] || 0);
    var montoReintNum = parseFloat(row[colMap["MontoReintegro"]] || 0);

    return {
      status: 'OK',
      idSolicitud: target,
      // Solicitante
      nombreSolicitante: (row[colMap["NombreSolicitante"]] || "").toString().trim(),
      correoSolicitante: (row[colMap["CorreoSolicitante"]] || "").toString().trim(),
      cargoSolicitante: (row[colMap["CargoSolicitante"]] || "").toString().trim(),
      gerencia: (row[colMap["Gerencia"]] || "").toString().trim(),
      centroCosto: (row[colMap["CentroCosto"]] || "").toString().trim(),
      agencia: (row[colMap["Agencia"]] || "").toString().trim(),
      // Viático
      duracionActividad: (row[colMap["DuracionActividad"]] || "Día único").toString().trim(),
      fechaInicio: (row[colMap["FechaInicio"]] || "").toString().trim(),
      fechaFin: (row[colMap["FechaFin"]] || "").toString().trim(),
      tipoViatico: (row[colMap["TipoViatico"]] || "").toString().trim(),
      horaEvento: (row[colMap["HoraEvento"]] || "").toString().trim(),
      tipoSolicitud: (row[colMap["TipoSolicitud"]] || "Personal").toString().trim(),
      destinatario: (row[colMap["Destinatario"]] || "").toString().trim(),
      monto: montoNum,
      montoFormatted: "$ " + montoNum.toFixed(2),
      clasificacionSolicitud: (row[colMap["ClasificacionSolicitud"]] || "Anticipo").toString().trim(),
      banco: (row[colMap["InstitucionBancaria"]] || "").toString().trim(),
      tipoCuenta: (row[colMap["TipoCuentaBancaria"]] || "").toString().trim(),
      numeroCuenta: (row[colMap["NumeroCuentaBancaria"]] || "").toString().trim(),
      motivoViatico: (row[colMap["MotivoViatico"]] || "").toString().trim(),
      esEditado: (row[colMap["EsEditado"]] || "No").toString().trim(),
      // Solicitud & Presupuesto
      fechaSolicitud: (row[colMap["FechaSolicitud"]] || "").toString().trim(),
      fechaModificacion: (row[colMap["FechaModificacion"]] || "").toString().trim(),
      rubroContable: (row[colMap["RubroContable"]] || "8110050400000 - Viáticos y Movilidad").toString().trim(),
      estadoSolicitud: (row[colMap["EstadoSolicitud"]] || "").toString().trim(),
      actorActual: (row[colMap["ActorActual"]] || "").toString().trim(),
      dentroPresupuesto: (row[colMap["DentroPresupuesto"]] || "Si").toString().trim(),
      justificacionPresupuesto: (row[colMap["JustificacionPresupuesto"]] || "").toString().trim(),
      archivosAdjuntos: archivos,
      // Autorizaciones
      datosAutorizacion: autorizaciones,
      // Provisión (E1)
      nombreProvision: (row[colMap["NombreProvision"]] || "").toString().trim(),
      fechaProvision: (row[colMap["FechaProvision"]] || "").toString().trim(),
      resolucionProvision: (row[colMap["ResolucionProvision"]] || "").toString().trim(),
      comentarioProvision: (row[colMap["ComentarioProvision"]] || "").toString().trim(),
      agrupableProvision: agrProv,
      // Procesamiento de Pago (E2)
      nombreProcesamiento: (row[colMap["NombreProcesamiento"]] || "").toString().trim(),
      fechaProcesamiento: (row[colMap["FechaProcesamiento"]] || "").toString().trim(),
      resolucionProcesamiento: (row[colMap["ResolucionProcesamiento"]] || "").toString().trim(),
      comentarioProcesamiento: (row[colMap["ComentarioProcesamiento"]] || "").toString().trim(),
      agrupableProcesamiento: agrProc,
      // Liquidación & Cierre (S3 / E3)
      tipoCierre: (row[colMap["TipoCierre"]] || "").toString().trim(),
      montoReintegro: montoReintNum,
      fechaReintegro: (row[colMap["FechaReintegro"]] || "").toString().trim(),
      fechaCierreS: (row[colMap["FechaCierreS"]] || "").toString().trim(),
      nombreCierreE: (row[colMap["NombreCierreE"]] || "").toString().trim(),
      fechaCierreE: (row[colMap["FechaCierreE"]] || "").toString().trim(),
      resolucionCierreE: (row[colMap["ResolucionCierreE"]] || "").toString().trim(),
      comentarioCierreE: (row[colMap["ComentarioCierreE"]] || "").toString().trim(),
      agrupableCierreE: agrCierre
    };
  } catch(err) {
    Logger.log("Error en obtenerExpedienteHistorico: " + err.toString());
    return { status: 'ERROR', mensaje: err.toString() };
  }
}
